Step 7 — Fix WebSocket URL in BrowserStreamEngine:
Find getRelayUrl or wherever the WebSocket URL is constructed. Replace with:
typescriptprivate getRelayUrl(): string {
  const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:'
  const host = window.location.hostname
  const port = window.location.port

  // On Replit production — same host no port needed
  if (!port || port === '443' || port === '80') {
    return `${protocol}//${host}/api/stream/ws`
  }

  // On Replit dev preview — use current port
  return `${protocol}//${host}:${port}/api/stream/ws`
}
