```typescript
startMediaRecorder(stream: MediaStream, config: StreamConfig) {
  // Use a LARGER chunk interval — smaller chunks (250ms) create more
  // WebSocket overhead and can cause out-of-order delivery issues
  const CHUNK_INTERVAL_MS = 1000 // 1 second chunks — more stable over WebSocket

  const mimeTypes = [
    'video/webm;codecs=h264,opus',  // h264 in webm is most FFmpeg-compatible
    'video/webm;codecs=vp9,opus',
    'video/webm;codecs=vp8,opus',
    'video/webm'
  ]

  let selectedMime = ''
  for (const mime of mimeTypes) {
    if (MediaRecorder.isTypeSupported(mime)) {
      selectedMime = mime
      console.log(`[StreamEngine] Using codec: ${mime}`)
      break
    }
  }

  this.mediaRecorder = new MediaRecorder(stream, {
    mimeType: selectedMime,
    videoBitsPerSecond: parseInt(config.videoBitrate) * 1000,
    audioBitsPerSecond: parseInt(config.audioBitrate) * 1000
  })

  // Buffer management — prevent WebSocket overload
  let pendingChunks: ArrayBuffer[] = []
  let isSending = false

  const sendNextChunk = async () => {
    if (isSending || pendingChunks.length === 0) return
    if (this.ws?.readyState !== WebSocket.OPEN) return

    isSending = true
    const chunk = pendingChunks.shift()!

    try {
      this.ws.send(chunk)
    } catch (e) {
      console.error('[StreamEngine] Send failed, requeueing:', e)
      pendingChunks.unshift(chunk) // put it back
    }

    isSending = false

    // Process next chunk if any
    if (pendingChunks.length > 0) {
      setTimeout(sendNextChunk, 10)
    }
  }

  this.mediaRecorder.ondataavailable = (event) => {
    if (event.data && event.data.size > 0) {
      event.data.arrayBuffer().then(buffer => {
        // Drop old chunks if queue gets too large — prevents memory buildup
        // that causes browser tab crashes during long streams
        if (pendingChunks.length > 5) {
          console.warn('[StreamEngine] Queue overflow — dropping oldest chunk')
          pendingChunks.shift()
        }
        pendingChunks.push(buffer)
        sendNextChunk()
      })
    }
  }

  this.mediaRecorder.onerror = (event: any) => {
    console.error('[StreamEngine] MediaRecorder error:', event.error)
    this.updateStats({ status: 'error', error: `Recording error: ${event.error?.message || 'unknown'}` })
  }

  this.mediaRecorder.start(CHUNK_INTERVAL_MS)
  console.log(`[StreamEngine] ✅ Recording started — ${CHUNK_INTERVAL_MS}ms chunks, codec: ${selectedMime}`)
}
```

```typescript
async connectToRelay(): Promise<void> {
  return new Promise((resolve, reject) => {
    const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:'
    const host = window.location.host // includes port if present
    const wsUrl = `${protocol}//${host}/api/stream/ws`

    console.log(`[StreamEngine] Connecting to: ${wsUrl}`)

          reject(new Error('FFmpeg not available on server — cannot stream'))
          return

        reject(new Error(`Server unreachable: ${err.message}. The deployment may be down or sleeping.`))
      })

    function attemptWebSocket() {
      const ws = new WebSocket(wsUrl)
      let resolved = false

      const timeout = setTimeout(() => {
        if (!resolved) {
          ws.close()
          reject(new Error(`WebSocket connection timeout after 10s to ${wsUrl}. Check that the WebSocket server is attached to the same HTTP server and path matches exactly.`))
        }

      ws.onopen = () => {
        console.log(`[StreamEngine] ✅ WebSocket connected: ${wsUrl}`)
      }

      ws.onmessage = (event) => {
        try {
          const msg = JSON.parse(event.data)
          if (msg.type === 'connected' && !resolved) {
            resolved = true
            clearTimeout(timeout)
            this.sessionId = msg.sessionId
            this.ws = ws
            this.setupWebSocketHandlers()
            resolve()
          }
          this.handleRelayMessage(msg)
        } catch {}
      }

      ws.onerror = (e) => {
        console.error('[StreamEngine] WebSocket error event:', e)
        if (!resolved) {

          reject(new Error(`WebSocket failed to connect to ${wsUrl}. This usually means: 1) the server is not running, 2) the deployment is Autoscale and sleeping, or 3) the WebSocket path is not registered on the server.`))
        }
      }

      ws.onclose = (e) => {
        console.log(`[StreamEngine] WebSocket closed: code=${e.code} reason=${e.reason}`)
        if (this.stats.status === 'live') {
          this.attemptReconnect(wsUrl)
        }
      }
    }
  })

// Auto-reconnect on unexpected disconnect during live stream
private reconnectAttempts = 0
private attemptReconnect(wsUrl: string) {
  if (this.reconnectAttempts >= 5) {
    this.updateStats({ status: 'error', error: 'Connection lost — max reconnect attempts reached' })
    return
  }
  this.reconnectAttempts++
  console.log(`[StreamEngine] Reconnect attempt ${this.reconnectAttempts}/5...`)

    ws.onopen = () => {
      console.log('[StreamEngine] ✅ Reconnected')
      this.reconnectAttempts = 0
      this.ws = ws
      this.setupWebSocketHandlers()
      // Re-send init config to resume FFmpeg with same RTMP target
      if (this.config) this.sendStreamConfig(this.config)
    }
    ws.onerror = () => this.attemptReconnect(wsUrl)
  }, 2000 * this.reconnectAttempts) // exponential backoff
}
```
