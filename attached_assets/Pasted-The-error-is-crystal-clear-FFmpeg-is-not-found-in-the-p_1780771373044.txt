const { execSync } = require('child_process')

// Resolve FFmpeg binary — tries multiple methods in order
function resolveFFmpegBinary() {
  const candidates = [
    // Environment variable set by replit.nix
    process.env.FFMPEG_PATH,
    // Common Nix store locations
    '/run/current-system/sw/bin/ffmpeg',
    '/nix/var/nix/profiles/default/bin/ffmpeg',
    // Try which command
    (() => { try { return execSync('which ffmpeg', { encoding: 'utf8' }).trim() } catch { return null } })(),
    // Try find in nix store
    (() => { try { return execSync('find /nix/store -name "ffmpeg" -type f 2>/dev/null | grep "bin/ffmpeg$" | head -1', { encoding: 'utf8' }).trim() } catch { return null } })(),
    // Fallback names
    'ffmpeg',
    '/usr/bin/ffmpeg',
    '/usr/local/bin/ffmpeg',
  ]

  for (const candidate of candidates) {
    if (!candidate) continue
    try {
      execSync(`${candidate} -version`, { encoding: 'utf8', timeout: 5000 })
      console.log(`[StreamRelay] ✅ FFmpeg resolved: ${candidate}`)
      return candidate
    } catch {}
  }

  console.error('[StreamRelay] ❌ FFmpeg not found — streaming will not work')
  return null
}

const FFMPEG_BIN = resolveFFmpegBinary()
