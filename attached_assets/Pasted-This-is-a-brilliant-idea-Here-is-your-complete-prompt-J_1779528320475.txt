**=== THE CORE ARCHITECTURE — READ THIS FIRST ===**

Here is exactly how the streaming pipeline works:

```
JJ NEXUS PRO webapp (full browser)
        ↓
getDisplayMedia() captures entire browser tab
        ↓
getUserMedia() captures microphone audio
        ↓
MediaRecorder encodes to webm chunks (real-time)
        ↓
WebSocket connection to a lightweight Node.js relay server
        ↓
Relay server converts webm chunks to RTMP using FFmpeg
        ↓
RTMP stream pushed to rtmp://live.restream.io/live/[KEY]
        ↓
Restream.io distributes to TikTok + YouTube + Instagram + all platforms
```

This is the exact same architecture used by browser-based streaming tools. It is fully achievable with free tools.

**=== PART 1: BUILD THE STREAMING RELAY SERVER ===**

Create a Node.js relay server that runs alongside the React app on Replit. This server receives video chunks from the browser via WebSocket and pipes them to FFmpeg which outputs RTMP.

Create `server/streamRelay.js`:

```javascript
// server/streamRelay.js
// Lightweight RTMP relay server
// Receives WebM chunks from browser → converts to RTMP via FFmpeg → pushes to Restream

const WebSocket = require('ws')
const { spawn } = require('child_process')
const http = require('http')
const path = require('path')

const HTTP_PORT = 3001 // WebSocket server port
const CORS_ORIGIN = '*' // Allow from any origin for Replit

// Active streaming sessions
const sessions = new Map()

// Create HTTP server for WebSocket upgrade
const server = http.createServer((req, res) => {
  // Health check endpoint
  if (req.url === '/health') {
    res.writeHead(200, { 'Content-Type': 'application/json' })
    res.end(JSON.stringify({
      status: 'running',
      activeSessions: sessions.size,
      timestamp: new Date().toISOString()
    }))
    return
  }

  // Stream status endpoint
  if (req.url === '/status') {
    const sessionList = Array.from(sessions.entries()).map(([id, session]) => ({
      id,
      rtmpUrl: session.rtmpUrl.replace(/\/[^/]+$/, '/***'), // hide key
      startTime: session.startTime,
      bytesReceived: session.bytesReceived,
      isLive: session.ffmpeg !== null
    }))
    res.writeHead(200, { 'Content-Type': 'application/json' })
    res.end(JSON.stringify({ sessions: sessionList }))
    return
  }

  res.writeHead(404)
  res.end()
})

// WebSocket server
const wss = new WebSocket.Server({ server })

wss.on('connection', (ws, req) => {
  const sessionId = `session_${Date.now()}_${Math.random().toString(36).slice(2)}`
  console.log(`[${sessionId}] New connection from ${req.socket.remoteAddress}`)

  const session = {
    id: sessionId,
    ws,
    ffmpeg: null,
    rtmpUrl: null,
    startTime: new Date(),
    bytesReceived: 0,
    chunksReceived: 0,
    isInitialized: false
  }

  sessions.set(sessionId, session)

  // Send session ID back to client
  ws.send(JSON.stringify({
    type: 'connected',
    sessionId,
    message: 'Stream relay ready'
  }))

  ws.on('message', (data) => {
    try {
      // First message is always JSON config
      if (!session.isInitialized) {
        const config = JSON.parse(data.toString())

        if (config.type === 'init') {
          console.log(`[${sessionId}] Initializing stream to: ${config.rtmpUrl.replace(/\/[^/]+$/, '/***')}`)
          session.rtmpUrl = config.rtmpUrl
          initializeFFmpeg(session, config)
        }
        return
      }

      // Subsequent messages are binary video chunks
      if (Buffer.isBuffer(data) || data instanceof ArrayBuffer) {
        const chunk = Buffer.isBuffer(data) ? data : Buffer.from(data)
        session.bytesReceived += chunk.length
        session.chunksReceived++

        if (session.ffmpeg && session.ffmpeg.stdin && !session.ffmpeg.stdin.destroyed) {
          const written = session.ffmpeg.stdin.write(chunk)
          if (!written) {
            // Backpressure — pause and wait for drain
            session.ffmpeg.stdin.once('drain', () => {})
          }
        }

        // Send stats back every 30 chunks
        if (session.chunksReceived % 30 === 0) {
          ws.send(JSON.stringify({
            type: 'stats',
            bytesReceived: session.bytesReceived,
            chunksReceived: session.chunksReceived,
            mbReceived: (session.bytesReceived / 1024 / 1024).toFixed(2),
            uptime: Math.floor((Date.now() - session.startTime.getTime()) / 1000)
          }))
        }
      }
    } catch (e) {
      console.error(`[${sessionId}] Message error:`, e.message)
    }
  })

  ws.on('close', (code, reason) => {
    console.log(`[${sessionId}] Connection closed: ${code} ${reason}`)
    stopSession(session)
    sessions.delete(sessionId)
  })

  ws.on('error', (err) => {
    console.error(`[${sessionId}] WebSocket error:`, err.message)
    stopSession(session)
    sessions.delete(sessionId)
  })
})

function initializeFFmpeg(session, config) {
  const {
    rtmpUrl,
    videoBitrate = '2500k',
    audioBitrate = '128k',
    fps = 30,
    resolution = '1280x720'
  } = config

  // FFmpeg command to convert WebM input to RTMP output
  // Input: WebM video from stdin (browser MediaRecorder output)
  // Output: RTMP stream to Restream/TikTok/YouTube
  const ffmpegArgs = [
    // Input configuration
    '-re',                          // Read at native frame rate
    '-i', 'pipe:0',                 // Read from stdin (WebSocket data)

    // Video encoding
    '-c:v', 'libx264',             // H.264 video codec (required for RTMP)
    '-preset', 'ultrafast',        // Fastest encoding (lowest CPU)
    '-tune', 'zerolatency',        // Minimize latency
    '-b:v', videoBitrate,          // Video bitrate
    '-maxrate', videoBitrate,      // Max bitrate
    '-bufsize', '2000k',           // Buffer size
    '-pix_fmt', 'yuv420p',        // Pixel format for compatibility
    '-g', String(fps * 2),         // Keyframe interval (2 seconds)
    '-r', String(fps),             // Output framerate

    // Audio encoding
    '-c:a', 'aac',                 // AAC audio codec (required for RTMP)
    '-b:a', audioBitrate,          // Audio bitrate
    '-ar', '44100',                // Audio sample rate
    '-ac', '2',                    // Stereo audio

    // Output
    '-f', 'flv',                   // FLV container (required for RTMP)
    rtmpUrl                        // RTMP destination URL
  ]

  console.log(`[${session.id}] Starting FFmpeg: ffmpeg ${ffmpegArgs.slice(-1)}`)

  session.ffmpeg = spawn('ffmpeg', ffmpegArgs, {
    stdio: ['pipe', 'pipe', 'pipe']
  })

  session.isInitialized = true

  session.ffmpeg.stdout.on('data', (data) => {
    // FFmpeg stdout (usually empty for RTMP output)
  })

  session.ffmpeg.stderr.on('data', (data) => {
    const log = data.toString()

    // Parse FFmpeg progress and send to client
    if (log.includes('fps=') || log.includes('bitrate=') || log.includes('speed=')) {
      const fpsMatch = log.match(/fps=\s*([\d.]+)/)
      const bitrateMatch = log.match(/bitrate=\s*([\d.]+kbits/)
      const speedMatch = log.match(/speed=\s*([\d.]+x)/)

      if (session.ws.readyState === 1) { // WebSocket.OPEN
        session.ws.send(JSON.stringify({
          type: 'ffmpeg_progress',
          fps: fpsMatch?.[1] || '0',
          bitrate: bitrateMatch?.[1] || '0kbits/s',
          speed: speedMatch?.[1] || '0x'
        }))
      }
    }

    // Log errors
    if (log.toLowerCase().includes('error') || log.toLowerCase().includes('failed')) {
      console.error(`[${session.id}] FFmpeg error:`, log.trim())

      if (session.ws.readyState === 1) {
        session.ws.send(JSON.stringify({
          type: 'error',
          message: `Stream error: ${log.trim().slice(0, 200)}`
        }))
      }
    }
  })

  session.ffmpeg.on('close', (code) => {
    console.log(`[${session.id}] FFmpeg exited with code ${code}`)

    if (session.ws.readyState === 1) {
      session.ws.send(JSON.stringify({
        type: 'stream_ended',
        code,
        message: code === 0 ? 'Stream ended cleanly' : `Stream ended with error code ${code}`
      }))
    }

    session.ffmpeg = null
  })

  session.ffmpeg.on('error', (err) => {
    console.error(`[${session.id}] FFmpeg spawn error:`, err.message)

    if (err.code === 'ENOENT') {
      session.ws.send(JSON.stringify({
        type: 'error',
        message: 'FFmpeg not installed on server. Run: apt-get install ffmpeg'
      }))
    }
  })

  // Notify client that stream is starting
  session.ws.send(JSON.stringify({
    type: 'stream_starting',
    message: 'FFmpeg initialized, stream starting...',
    rtmpDestination: rtmpUrl.replace(/\/[^/]+$/, '/***')
  }))

  console.log(`[${session.id}] ✅ FFmpeg started, streaming to RTMP`)
}

function stopSession(session) {
  if (session.ffmpeg) {
    try {
      session.ffmpeg.stdin.end()
      setTimeout(() => {
        if (session.ffmpeg) {
          session.ffmpeg.kill('SIGTERM')
        }
      }, 2000)
    } catch (e) {}
    session.ffmpeg = null
  }
}

// Graceful shutdown
process.on('SIGTERM', () => {
  console.log('Shutting down relay server...')
  sessions.forEach(session => stopSession(session))
  server.close(() => process.exit(0))
})

server.listen(HTTP_PORT, '0.0.0.0', () => {
  console.log(`✅ JJ NEXUS PRO Stream Relay Server running on port ${HTTP_PORT}`)
  console.log(`📡 WebSocket endpoint: ws://localhost:${HTTP_PORT}`)
  console.log(`🔍 Health check: http://localhost:${HTTP_PORT}/health`)
})

module.exports = { server, wss }
```

**=== PART 2: BROWSER STREAMING ENGINE ===**

Create `src/services/BrowserStreamEngine.ts` — the core client-side streaming service:

```typescript
// src/services/BrowserStreamEngine.ts
// Captures the full browser screen and streams to the relay server

export interface StreamConfig {
  rtmpUrl: string
  streamKey: string
  resolution: '1280x720' | '1920x1080' | '854x480'
  fps: 30 | 60 | 24
  videoBitrate: '1500k' | '2500k' | '4000k' | '6000k'
  audioBitrate: '128k' | '192k'
  captureMode: 'full_tab' | 'full_screen' | 'window'
  includeMicrophone: boolean
  includeSystemAudio: boolean
}

export interface StreamStats {
  status: 'idle' | 'connecting' | 'live' | 'error' | 'stopped'
  duration: number // seconds
  fps: number
  bitrate: string
  bytesSent: number
  mbSent: number
  chunksSeent: number
  viewers: number
  error: string | null
  startTime: Date | null
  ffmpegFps: string
  ffmpegBitrate: string
  ffmpegSpeed: string
}

type StatsCallback = (stats: StreamStats) => void

export class BrowserStreamEngine {
  private ws: WebSocket | null = null
  private mediaRecorder: MediaRecorder | null = null
  private displayStream: MediaStream | null = null
  private micStream: MediaStream | null = null
  private combinedStream: MediaStream | null = null
  private sessionId: string | null = null
  private statsTimer: NodeJS.Timeout | null = null
  private durationTimer: NodeJS.Timeout | null = null
  private onStatsUpdate: StatsCallback | null = null
  private config: StreamConfig | null = null

  private stats: StreamStats = {
    status: 'idle',
    duration: 0,
    fps: 0,
    bitrate: '0 kbps',
    bytesSent: 0,
    mbSent: 0,
    chunksSent: 0,
    viewers: 0,
    error: null,
    startTime: null,
    ffmpegFps: '0',
    ffmpegBitrate: '0',
    ffmpegSpeed: '0x'
  }

  onStats(callback: StatsCallback) {
    this.onStatsUpdate = callback
  }

  private updateStats(partial: Partial<StreamStats>) {
    this.stats = { ...this.stats, ...partial }
    this.onStatsUpdate?.(this.stats)
  }

  // Get relay server WebSocket URL
  private getRelayUrl(): string {
    const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:'
    const host = window.location.hostname
    // In Replit, the relay server runs on port 3001
    return `${protocol}//${host}:3001`
  }

  // Step 1: Capture the full browser tab or screen
  async captureScreen(config: StreamConfig): Promise<MediaStream> {
    const [width, height] = config.resolution.split('x').map(Number)

    let displayStream: MediaStream

    if (config.captureMode === 'full_tab') {
      // Capture entire browser tab — shows the full JJ NEXUS PRO webapp
      displayStream = await navigator.mediaDevices.getDisplayMedia({
        video: {
          width: { ideal: width, max: 1920 },
          height: { ideal: height, max: 1080 },
          frameRate: { ideal: config.fps, max: 60 },
          cursor: 'always', // show cursor
          displaySurface: 'browser' // prefer browser tab
        } as any,
        audio: config.includeSystemAudio ? {
          echoCancellation: false,
          noiseSuppression: false,
          sampleRate: 44100
        } : false,
        preferCurrentTab: true // Chrome 94+ — prefer the current tab
      } as any)
    } else if (config.captureMode === 'full_screen') {
      // Capture entire monitor
      displayStream = await navigator.mediaDevices.getDisplayMedia({
        video: {
          width: { ideal: width },
          height: { ideal: height },
          frameRate: { ideal: config.fps },
          displaySurface: 'monitor'
        } as any,
        audio: config.includeSystemAudio
      })
    } else {
      // Capture specific window
      displayStream = await navigator.mediaDevices.getDisplayMedia({
        video: {
          width: { ideal: width },
          height: { ideal: height },
          frameRate: { ideal: config.fps },
          displaySurface: 'window'
        } as any,
        audio: config.includeSystemAudio
      })
    }

    this.displayStream = displayStream

    // Handle screen share stop (user clicks "Stop sharing" browser button)
    displayStream.getVideoTracks()[0].addEventListener('ended', () => {
      console.log('Screen share stopped by user')
      this.stopStream()
    })

    return displayStream
  }

  // Step 2: Capture microphone audio
  async captureMicrophone(): Promise<MediaStream | null> {
    try {
      const micStream = await navigator.mediaDevices.getUserMedia({
        audio: {
          echoCancellation: true,
          noiseSuppression: true,
          autoGainControl: true,
          sampleRate: 44100,
          channelCount: 2
        },
        video: false
      })
      this.micStream = micStream
      return micStream
    } catch (e) {
      console.warn('Microphone not available:', e)
      return null
    }
  }

  // Step 3: Combine video + audio tracks into one stream
  combineStreams(displayStream: MediaStream, micStream: MediaStream | null): MediaStream {
    const tracks: MediaStreamTrack[] = []

    // Add video track from screen capture
    displayStream.getVideoTracks().forEach(track => tracks.push(track))

    if (micStream && displayStream.getAudioTracks().length > 0) {
      // Mix system audio + microphone using AudioContext
      const audioCtx = new AudioContext()
      const dest = audioCtx.createMediaStreamDestination()

      // System audio source
      const sysSource = audioCtx.createMediaStreamSource(
        new MediaStream(displayStream.getAudioTracks())
      )
      const sysGain = audioCtx.createGain()
      sysGain.gain.value = 1.0
      sysSource.connect(sysGain)
      sysGain.connect(dest)

      // Microphone source
      const micSource = audioCtx.createMediaStreamSource(micStream)
      const micGain = audioCtx.createGain()
      micGain.value = 1.0
      micSource.connect(micGain)
      micGain.connect(dest)

      // Add mixed audio track
      dest.stream.getAudioTracks().forEach(track => tracks.push(track))
    } else if (micStream) {
      // Only microphone audio
      micStream.getAudioTracks().forEach(track => tracks.push(track))
    } else if (displayStream.getAudioTracks().length > 0) {
      // Only system audio
      displayStream.getAudioTracks().forEach(track => tracks.push(track))
    }

    const combined = new MediaStream(tracks)
    this.combinedStream = combined
    return combined
  }

  // Step 4: Connect to relay server via WebSocket
  async connectToRelay(): Promise<void> {
    return new Promise((resolve, reject) => {
      const url = this.getRelayUrl()
      console.log('Connecting to relay server:', url)
      this.updateStats({ status: 'connecting' })

      this.ws = new WebSocket(url)

      this.ws.onopen = () => {
        console.log('✅ Connected to relay server')
      }

      this.ws.onmessage = (event) => {
        try {
          const msg = JSON.parse(event.data)

          switch (msg.type) {
            case 'connected':
              this.sessionId = msg.sessionId
              console.log('Session ID:', this.sessionId)
              resolve()
              break

            case 'stream_starting':
              console.log('Stream starting:', msg.message)
              this.updateStats({ status: 'live', startTime: new Date() })
              this.startDurationTimer()
              break

            case 'ffmpeg_progress':
              this.updateStats({
                ffmpegFps: msg.fps,
                ffmpegBitrate: msg.bitrate,
                ffmpegSpeed: msg.speed
              })
              break

            case 'stats':
              this.updateStats({
                bytesSent: msg.bytesReceived,
                mbSent: parseFloat(msg.mbReceived)
              })
              break

            case 'error':
              console.error('Relay error:', msg.message)
              this.updateStats({ status: 'error', error: msg.message })
              break

            case 'stream_ended':
              this.updateStats({ status: 'stopped' })
              break
          }
        } catch (e) {}
      }

      this.ws.onerror = (error) => {
        console.error('WebSocket error:', error)
        this.updateStats({
          status: 'error',
          error: 'Cannot connect to stream relay server. Make sure the server is running.'
        })
        reject(new Error('WebSocket connection failed'))
      }

      this.ws.onclose = (event) => {
        console.log('WebSocket closed:', event.code, event.reason)
        if (this.stats.status === 'live') {
          this.updateStats({ status: 'stopped' })
        }
      }

      // Timeout after 10 seconds
      setTimeout(() => {
        if (this.stats.status === 'connecting') {
          reject(new Error('Connection timeout — relay server not responding'))
        }
      }, 10000)
    })
  }

  // Step 5: Send stream config to relay server
  sendStreamConfig(config: StreamConfig) {
    if (!this.ws || this.ws.readyState !== WebSocket.OPEN) return

    const fullRtmpUrl = `${config.rtmpUrl}/${config.streamKey}`

    this.ws.send(JSON.stringify({
      type: 'init',
      rtmpUrl: fullRtmpUrl,
      videoBitrate: config.videoBitrate,
      audioBitrate: config.audioBitrate,
      fps: config.fps,
      resolution: config.resolution
    }))

    console.log('Stream config sent to relay')
  }

  // Step 6: Start MediaRecorder and pipe chunks to WebSocket
  startMediaRecorder(stream: MediaStream, config: StreamConfig) {
    // Select best supported MIME type
    const mimeTypes = [
      'video/webm;codecs=h264,opus',
      'video/webm;codecs=vp9,opus',
      'video/webm;codecs=vp8,opus',
      'video/webm',
    ]

    let selectedMime = ''
    for (const mime of mimeTypes) {
      if (MediaRecorder.isTypeSupported(mime)) {
        selectedMime = mime
        break
      }
    }

    if (!selectedMime) {
      throw new Error('No supported video codec found in this browser')
    }

    console.log('Using MIME type:', selectedMime)

    const bitrateNum = parseInt(config.videoBitrate) * 1000

    this.mediaRecorder = new MediaRecorder(stream, {
      mimeType: selectedMime,
      videoBitsPerSecond: bitrateNum,
      audioBitsPerSecond: parseInt(config.audioBitrate) * 1000
    })

    let chunkCount = 0
    let totalBytes = 0
    let lastBitrateCalc = Date.now()
    let bytesSinceLastCalc = 0

    this.mediaRecorder.ondataavailable = (event) => {
      if (event.data && event.data.size > 0 && this.ws?.readyState === WebSocket.OPEN) {
        event.data.arrayBuffer().then(buffer => {
          if (this.ws?.readyState === WebSocket.OPEN) {
            this.ws.send(buffer)
            chunkCount++
            totalBytes += buffer.byteLength
            bytesSinceLastCalc += buffer.byteLength

            // Calculate real-time bitrate
            const now = Date.now()
            if (now - lastBitrateCalc >= 1000) {
              const bps = (bytesSinceLastCalc * 8) / ((now - lastBitrateCalc) / 1000)
              const kbps = Math.round(bps / 1000)
              this.updateStats({
                bitrate: `${kbps} kbps`,
                bytesSent: totalBytes,
                mbSent: parseFloat((totalBytes / 1024 / 1024).toFixed(2)),
                chunksSent: chunkCount
              })
              bytesSinceLastCalc = 0
              lastBitrateCalc = now
            }
          }
        })
      }
    }

    this.mediaRecorder.onerror = (event) => {
      console.error('MediaRecorder error:', event)
      this.updateStats({ status: 'error', error: 'Recording error occurred' })
    }

    // Collect chunks every 250ms for low latency
    this.mediaRecorder.start(250)
    console.log('✅ MediaRecorder started')
  }

  // Duration timer
  startDurationTimer() {
    this.durationTimer = setInterval(() => {
      this.updateStats({ duration: this.stats.duration + 1 })
    }, 1000)
  }

  // MAIN: Start the full streaming pipeline
  async startStream(config: StreamConfig): Promise<void> {
    this.config = config
    this.updateStats({ status: 'connecting', error: null })

    try {
      // Step 1: Capture screen
      console.log('Step 1: Capturing screen...')
      this.updateStats({ status: 'connecting' })
      const displayStream = await this.captureScreen(config)

      // Step 2: Capture microphone (optional)
      let micStream: MediaStream | null = null
      if (config.includeMicrophone) {
        console.log('Step 2: Capturing microphone...')
        micStream = await this.captureMicrophone()
      }

      // Step 3: Combine streams
      console.log('Step 3: Combining streams...')
      const combinedStream = this.combineStreams(displayStream, micStream)

      // Step 4: Connect to relay server
      console.log('Step 4: Connecting to relay server...')
      await this.connectToRelay()

      // Step 5: Send config to relay
      console.log('Step 5: Sending stream config...')
      this.sendStreamConfig(config)

      // Step 6: Start recording and piping to WebSocket
      console.log('Step 6: Starting MediaRecorder...')
      await new Promise(resolve => setTimeout(resolve, 500)) // small delay
      this.startMediaRecorder(combinedStream, config)

      console.log('✅ Full streaming pipeline started')

    } catch (error: any) {
      console.error('Stream start failed:', error)
      this.updateStats({
        status: 'error',
        error: error.message || 'Failed to start stream'
      })
      this.cleanup()
      throw error
    }
  }

  // Stop everything cleanly
  stopStream() {
    console.log('Stopping stream...')

    if (this.mediaRecorder && this.mediaRecorder.state !== 'inactive') {
      this.mediaRecorder.stop()
    }

    if (this.displayStream) {
      this.displayStream.getTracks().forEach(t => t.stop())
    }

    if (this.micStream) {
      this.micStream.getTracks().forEach(t => t.stop())
    }

    if (this.ws) {
      this.ws.close(1000, 'Stream stopped by user')
    }

    if (this.durationTimer) clearInterval(this.durationTimer)
    if (this.statsTimer) clearInterval(this.statsTimer)

    this.cleanup()
    this.updateStats({ status: 'stopped' })
    console.log('✅ Stream stopped')
  }

  private cleanup() {
    this.mediaRecorder = null
    this.displayStream = null
    this.micStream = null
    this.combinedStream = null
    this.ws = null
    this.sessionId = null
  }

  // Format duration as HH:MM:SS
  static formatDuration(seconds: number): string {
    const h = Math.floor(seconds / 3600)
    const m = Math.floor((seconds % 3600) / 60)
    const s = seconds % 60
    return [h, m, s].map(v => String(v).padStart(2, '0')).join(':')
  }
}

export const streamEngine = new BrowserStreamEngine()
```
