**JJ NEXUS PRO — PRODUCTION STREAMING FIX: WEBSOCKET + FFMPEG UNIFIED SERVER**

The streaming works perfectly in development but crashes after publishing. This is a known production deployment problem. Read this entire prompt carefully before touching any code.

---

**=== THE ROOT CAUSE ===**

In development Vite proxies the WebSocket from the frontend to the API server. In production there is no Vite and no proxy. The published app serves only static files. When the browser tries to connect to the WebSocket relay it hits a dead static file server and gets rejected. FFmpeg never starts. Stream never goes live.

The fix is to make one single Express server that does everything on one port: serves the static frontend files, handles all API routes, and handles the WebSocket relay. Then the deployment runs only that one server and everything works.

```typescript
import express from 'express'
import { createServer } from 'http'
import { WebSocketServer, WebSocket } from 'ws'
import { spawn, ChildProcess } from 'child_process'
import path from 'path'
import { existsSync } from 'fs'
import https from 'https'
import http from 'http'

const app = express()
const PORT = parseInt(process.env.PORT || '5000', 10)
const IS_PRODUCTION = process.env.NODE_ENV === 'production'

app.use(express.json())

// ═══════════════════════════════════════
// CORS — allow everything
// ═══════════════════════════════════════
app.use((req, res, next) => {
  res.setHeader('Access-Control-Allow-Origin', '*')
  res.setHeader('Access-Control-Allow-Methods', 'GET, POST, OPTIONS')
  res.setHeader('Access-Control-Allow-Headers', 'Content-Type')
  if (req.method === 'OPTIONS') { res.status(200).end(); return }
  next()
})

// ═══════════════════════════════════════
// PROXY HELPER — routes all external API calls through server
// This fixes ALL CORS errors in the browser
// ═══════════════════════════════════════
function proxyRequest(targetUrl: string, res: express.Response) {
  const url = new URL(targetUrl)
  const protocol = url.protocol === 'https:' ? https : http

  const options = {
    hostname: url.hostname,
    path: url.pathname + url.search,
    method: 'GET',
    headers: {
      'User-Agent': 'Mozilla/5.0 (compatible; JJNexusPro/1.0)',
      'Accept': 'application/json',
    },
    timeout: 12000
  }

  const proxyReq = protocol.request(options, (proxyRes) => {
    let data = ''
    proxyRes.on('data', (chunk: Buffer) => data += chunk.toString())
    proxyRes.on('end', () => {
      res.setHeader('Content-Type', 'application/json')
      res.setHeader('Access-Control-Allow-Origin', '*')
      res.status(200).send(data)
    })
  })

  proxyReq.on('error', (err: Error) => {
    res.status(502).json({ error: err.message })
  })

  proxyReq.on('timeout', () => {
    proxyReq.destroy()
    res.status(504).json({ error: 'Upstream timeout' })
  })

  proxyReq.end()
}

// ═══════════════════════════════════════
// PROXY ROUTES — ALL external APIs go through here
// ═══════════════════════════════════════
app.get('/api/stream/proxy/forex-rates', (req, res) => {

app.get('/api/stream/proxy/gold-price', (req, res) => {

// ═══════════════════════════════════════
// WEBSOCKET RELAY — stream chunks to FFmpeg → RTMP
// ═══════════════════════════════════════
interface StreamSession {
  ws: WebSocket
  ffmpeg: ChildProcess | null
  rtmpUrl: string
  startTime: Date
  bytesReceived: number
  isInitialized: boolean
}

const sessions = new Map<string, StreamSession>()

// ═══════════════════════════════════════
// STATIC FRONTEND — production only
// ═══════════════════════════════════════

// Find the correct dist path
const POSSIBLE_DIST_PATHS = [
  path.join(process.cwd(), 'dist'),
  path.join(process.cwd(), 'dist', 'public'),
  path.join(process.cwd(), 'ethiostream-pro', 'dist'),
  path.join(process.cwd(), 'ethiostream-pro', 'dist', 'public'),
  path.join(process.cwd(), '..', 'dist'),
  path.join(process.cwd(), 'public'),
]

let DIST_PATH = ''
for (const p of POSSIBLE_DIST_PATHS) {
  if (existsSync(path.join(p, 'index.html'))) {
    DIST_PATH = p
    console.log(`✅ Found frontend build at: ${p}`)
    break
  }
}

if (IS_PRODUCTION) {
  if (DIST_PATH) {
    // Serve static assets
    app.use(express.static(DIST_PATH))
    // SPA fallback — all non-API routes return index.html
    app.get('/{*splat}', (req, res) => {
      if (req.path.startsWith('/api/')) {
        res.status(404).json({ error: 'API route not found' })
        return
      }
      res.sendFile(path.join(DIST_PATH, 'index.html'))
    })
    console.log(`✅ Serving static frontend from: ${DIST_PATH}`)
  } else {
    console.error('❌ Frontend build not found. Run: npm run build')
    // Still serve API even without frontend
    app.get('/', (req, res) => {
      res.json({ message: 'JJ NEXUS PRO API running. Frontend not built.' })
    })
  }
}

// ═══════════════════════════════════════
// CREATE HTTP SERVER
// ═══════════════════════════════════════
const httpServer = createServer(app)

// ═══════════════════════════════════════
// WEBSOCKET SERVER — attached to HTTP server
// ═══════════════════════════════════════
const wss = new WebSocketServer({
  server: httpServer,
  path: '/api/stream/ws'
})

wss.on('connection', (ws: WebSocket) => {
  const sessionId = `sess_${Date.now()}_${Math.random().toString(36).slice(2)}`
  console.log(`[${sessionId}] New WebSocket connection`)

  const session: StreamSession = {
    ws,
    ffmpeg: null,
    rtmpUrl: '',
    startTime: new Date(),
    bytesReceived: 0,
    isInitialized: false
  }

  sessions.set(sessionId, session)

  ws.send(JSON.stringify({
    type: 'connected',
    sessionId,
    message: 'JJ NEXUS PRO Stream Relay ready'
  }))

  ws.on('message', (data: Buffer | string) => {
    try {
      if (!session.isInitialized) {
        const config = JSON.parse(data.toString())
        if (config.type === 'init') {
          console.log(`[${sessionId}] Init stream to RTMP`)
          session.rtmpUrl = config.rtmpUrl
          startFFmpeg(session, sessionId, config)
        }
        return
      }

      if (Buffer.isBuffer(data) || data instanceof Buffer) {
        session.bytesReceived += data.length
        if (session.ffmpeg?.stdin && !session.ffmpeg.stdin.destroyed) {
          session.ffmpeg.stdin.write(data)
        }

        if (Math.floor(session.bytesReceived / (1024 * 256)) > Math.floor((session.bytesReceived - data.length) / (1024 * 256))) {
          ws.send(JSON.stringify({
            type: 'stats',
            bytesReceived: session.bytesReceived,
            mbReceived: (session.bytesReceived / 1024 / 1024).toFixed(2),
            uptime: Math.floor((Date.now() - session.startTime.getTime()) / 1000)
          }))
        }
      }
    } catch (e) {
      console.error(`[${sessionId}] Message error:`, e)
    }
  })

  ws.on('close', () => {
    console.log(`[${sessionId}] Connection closed`)
    stopSession(session)
    sessions.delete(sessionId)
  })

  ws.on('error', (err) => {
    console.error(`[${sessionId}] WebSocket error:`, err.message)
    stopSession(session)
    sessions.delete(sessionId)
  })
})

function startFFmpeg(session: StreamSession, sessionId: string, config: any) {
  const {
    rtmpUrl,
    videoBitrate = '2500k',
    audioBitrate = '128k',
    fps = 30
  } = config

  const ffmpegArgs = [
    '-re',
    '-i', 'pipe:0',
    '-c:v', 'libx264',
    '-preset', 'ultrafast',
    '-tune', 'zerolatency',
    '-b:v', videoBitrate,
    '-maxrate', videoBitrate,
    '-bufsize', '2000k',
    '-pix_fmt', 'yuv420p',
    '-g', String(fps * 2),
    '-r', String(fps),
    '-c:a', 'aac',
    '-b:a', audioBitrate,
    '-ar', '44100',
    '-ac', '2',
    '-f', 'flv',
    rtmpUrl
  ]

  console.log(`[${sessionId}] Starting FFmpeg → ${rtmpUrl.replace(/\/[^/]+$/, '/***')}`)

  session.ffmpeg = spawn('ffmpeg', ffmpegArgs, { stdio: ['pipe', 'pipe', 'pipe'] })
  session.isInitialized = true

  session.ffmpeg.stderr?.on('data', (data: Buffer) => {
    const log = data.toString()
    const fpsMatch = log.match(/fps=\s*([\d.]+)/)
    const bitrateMatch = log.match(/bitrate=\s*([\d.]+kbits\/s)/)
    const speedMatch = log.match(/speed=\s*([\d.]+x)/)

    if (fpsMatch || bitrateMatch) {
      try {
        session.ws.send(JSON.stringify({
          type: 'ffmpeg_progress',
          fps: fpsMatch?.[1] || '0',
          bitrate: bitrateMatch?.[1] || '0kbits/s',
          speed: speedMatch?.[1] || '0x'
        }))
      } catch {}
    }

    if (log.toLowerCase().includes('error') && !log.includes('frame=')) {
      console.error(`[${sessionId}] FFmpeg:`, log.trim().slice(0, 200))
    }
  })

  session.ffmpeg.on('close', (code: number) => {
    console.log(`[${sessionId}] FFmpeg exited: ${code}`)
    try {
      session.ws.send(JSON.stringify({
        type: code === 0 ? 'stream_ended' : 'error',
        message: code === 0 ? 'Stream ended' : `FFmpeg exited with code ${code}`
      }))
    } catch {}
    session.ffmpeg = null
  })

  session.ffmpeg.on('error', (err: Error) => {
    console.error(`[${sessionId}] FFmpeg error:`, err.message)
    try {
      session.ws.send(JSON.stringify({
        type: 'error',
        message: err.message.includes('ENOENT')
          ? 'FFmpeg not found. Make sure ffmpeg-full is in replit.nix'
          : err.message
      }))
    } catch {}
  })

  session.ws.send(JSON.stringify({
    type: 'stream_starting',
    message: 'FFmpeg started — stream is live'
  }))
}

function stopSession(session: StreamSession) {
  if (session.ffmpeg) {
    try {
      session.ffmpeg.stdin?.end()
      setTimeout(() => { session.ffmpeg?.kill('SIGTERM') }, 2000)
    } catch {}
    session.ffmpeg = null
  }
}

// ═══════════════════════════════════════
// START SERVER
// ═══════════════════════════════════════
httpServer.listen(PORT, '0.0.0.0', () => {
  console.log(`✅ JJ NEXUS PRO unified server running on port ${PORT}`)
  console.log(`📡 WebSocket relay: ws://localhost:${PORT}/api/stream/ws`)
  console.log(`🔍 Health: http://localhost:${PORT}/api/stream/health`)
  console.log(`🌍 Environment: ${IS_PRODUCTION ? 'PRODUCTION' : 'development'}`)
  if (DIST_PATH) console.log(`📁 Frontend: ${DIST_PATH}`)
})

process.on('SIGTERM', () => {
  console.log('Shutting down...')
  sessions.forEach(stopSession)
  httpServer.close(() => process.exit(0))
})

export default app
```

```typescript
private getRelayUrl(): string {
  const host = window.location.hostname
  const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:'

  // In production the WebSocket is on the SAME host and port as the app
  // No proxy needed — Express handles it directly
  if (window.location.port && window.location.port !== '443' && window.location.port !== '80') {
    // Development: specific port
    return `${protocol}//${host}:${window.location.port}/api/stream/ws`
  }

  // Production: same host, default port, correct path
  return `${protocol}//${host}/api/stream/ws`
}
```
