```typescript
      const errData = await response.json().catch(() => ({}))
      throw new Error(errData?.error?.message || `HTTP ${response.status}`)

    const data = await response.json()

    if (!data || !data.content || !Array.isArray(data.content)) {
      throw new Error('Invalid response structure from API')
    }

    const textBlocks = data.content.filter((b: any) => b.type === 'text')
    if (textBlocks.length === 0) {
      // Model may have only returned tool use blocks — try to get tool results
      return 'Analysis completed. The AI searched for live data. Please try again for the full response.'
    }

    return textBlocks.map((b: any) => b.text).join('\n')

    throw new Error(`Alchemist AI error: ${err.message}`)
  }
}
```

```typescript
interface Props {
  symbol?: string
  style?: React.CSSProperties
}

export default function TradingViewWidget({ symbol = 'XAUUSD', style }: Props) {
  const containerRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    if (!containerRef.current) return
    containerRef.current.innerHTML = ''
    const script = document.createElement('script')
    script.src = 'https://s3.tradingview.com/external-embedding/embed-widget-advanced-chart.js'
    script.type = 'text/javascript'
    script.async = true
    script.innerHTML = JSON.stringify({
      symbol: `FX:${symbol}`,
      interval: 'H1',
      timezone: 'Etc/UTC',
      theme: 'dark',
      style: '1',
      locale: 'en',
      toolbar_bg: '#0a0a0a',
      enable_publishing: false,
      allow_symbol_change: true,
      container_id: 'jjnexus_chart',
      height: '100%',
      width: '100%',
      studies: ['RSI@tv-basicstudies', 'MACD@tv-basicstudies', 'Volume@tv-basicstudies'],
    })
    containerRef.current.appendChild(script)
  }, [symbol])

  return (
    <div ref={containerRef} id="jjnexus_chart"
      style={{ width: '100%', height: '100%', ...style }} />
  )
}
```

**=== FIX 4 — REAL-TIME ECONOMIC ALERTS & NOTIFICATIONS ===**

Create `src/services/AlertService.ts`:

```typescript
export class AlertService {
  private static priceAlerts: Map<string, { price: number; direction: 'above' | 'below'; pair: string }> = new Map()
  private static eventAlerts: Map<string, { minutesBefore: number; eventTitle: string }> = new Map()
  private static notificationsEnabled = false

  static async requestPermission(): Promise<boolean> {
    if (!('Notification' in window)) return false
    const permission = await Notification.requestPermission()
    this.notificationsEnabled = permission === 'granted'
    return this.notificationsEnabled
  }

  static notify(title: string, body: string, icon = '/logo.png') {
    if (this.notificationsEnabled && Notification.permission === 'granted') {
      new Notification(title, { body, icon })
    }
    // Also show in-app toast notification
    const event = new CustomEvent('jjnexus-notification', {
      detail: { title, body, timestamp: Date.now() }
    })
    window.dispatchEvent(event)
  }

  static addPriceAlert(pair: string, targetPrice: number, direction: 'above' | 'below') {
    const id = `${pair}-${direction}-${targetPrice}`
    this.priceAlerts.set(id, { price: targetPrice, direction, pair })
    this.notify('✅ Alert Set', `${pair} alert set for ${direction} ${targetPrice}`)
  }

  static checkPriceAlerts(currentPrices: Record<string, number>) {
    this.priceAlerts.forEach((alert, id) => {
      const currentPrice = currentPrices[alert.pair]
      if (!currentPrice) return
      const triggered =
        alert.direction === 'above' ? currentPrice >= alert.price :
        currentPrice <= alert.price
      if (triggered) {
        this.notify(
          `🚨 ${alert.pair} Alert Triggered!`,
          `Price is now ${alert.direction} ${alert.price}. Current: ${currentPrice.toFixed(5)}`
        )
        this.priceAlerts.delete(id)
      }
    })
  }

  static scheduleEventAlert(eventId: string, eventDate: string, eventTitle: string, minutesBefore: number) {
    const alertTime = new Date(eventDate).getTime() - minutesBefore * 60 * 1000
    const now = Date.now()
    if (alertTime <= now) return
    const delay = alertTime - now
    setTimeout(() => {
      this.notify(
        `⏰ Upcoming: ${eventTitle}`,
        `${eventTitle} starts in ${minutesBefore} minutes! Check the economic calendar.`
      )
    }, delay)
    this.eventAlerts.set(eventId, { minutesBefore, eventTitle })
  }
}
```

```typescript
// TikTok Live Chat: Use TikTok Live Center webhook or tiktok-live-connector
// YouTube Live Chat: Use YouTube Data API v3 liveChatMessages endpoint

export class LiveChatService {
  private pollInterval: NodeJS.Timeout | null = null
  private chatMessages: ChatMessage[] = []
  private pinnedMessages: string[] = []

  // YouTube Live Chat (requires OAuth + live broadcast ID)
  async connectYouTube(broadcastId: string, apiKey: string) {
    const fetchMessages = async () => {
      try {
        const res = await fetch(
          `https://www.googleapis.com/youtube/v3/liveChat/messages?liveChatId=${broadcastId}&part=snippet,authorDetails&key=${apiKey}`
        )
        const data = await res.json()
        const messages = data.items?.map((item: any) => ({
          id: item.id,
          author: item.authorDetails.displayName,
          text: item.snippet.displayMessage,
          avatar: item.authorDetails.profileImageUrl,
          timestamp: item.snippet.publishedAt,
          platform: 'youtube' as const,
          pinned: this.pinnedMessages.includes(item.id)
        })) || []
        return messages
      } catch (e) {
        console.error('YouTube chat fetch failed:', e)
        return []
      }
    }
    this.pollInterval = setInterval(async () => {
      const msgs = await fetchMessages()
      // dispatch to UI
      window.dispatchEvent(new CustomEvent('chat-messages', { detail: msgs }))
    }, 5000)
  }

  // TikTok Live Chat: requires tiktok-live-connector server-side
  // For client-side, use a WebSocket proxy server
  connectTikTok(username: string, wsServerUrl: string) {
    const ws = new WebSocket(wsServerUrl)
    ws.onmessage = (event) => {
      const msg = JSON.parse(event.data)
      window.dispatchEvent(new CustomEvent('chat-messages', {
        detail: [{ ...msg, platform: 'tiktok' }]
      }))
    }
  }

  pinMessage(messageId: string) {
    this.pinnedMessages.push(messageId)
  }

  unpinMessage(messageId: string) {
    this.pinnedMessages = this.pinnedMessages.filter(id => id !== messageId)
  }

  disconnect() {
    if (this.pollInterval) clearInterval(this.pollInterval)
  }
}
```

```typescript
// src/hooks/useStreamStatus.ts
import { useState, useEffect, useRef } from 'react'
import { useCamera } from '../context/CameraContext'

export function useStreamStatus() {
  const { stream } = useCamera()
  const [fps, setFps] = useState(0)
  const [resolution, setResolution] = useState('No signal')
  const [bitrate, setBitrate] = useState('0 Mbps')
  const [status, setStatus] = useState<'idle' | 'live' | 'recording' | 'error'>('idle')
  const frameCount = useRef(0)
  const lastTime = useRef(performance.now())
  const lastBytes = useRef(0)

  // Real FPS from requestAnimationFrame
  useEffect(() => {
    if (!stream) { setFps(0); return }
    let rafId: number
    const measure = () => {
      frameCount.current++
      const now = performance.now()
      if (now - lastTime.current >= 1000) {
        setFps(frameCount.current)
        frameCount.current = 0
        lastTime.current = now
      }
      rafId = requestAnimationFrame(measure)
    }
    rafId = requestAnimationFrame(measure)
    return () => cancelAnimationFrame(rafId)
  }, [stream])

  // Real resolution from stream track settings
  useEffect(() => {
    if (!stream) { setResolution('No signal'); return }
    const track = stream.getVideoTracks()[0]
    if (!track) { setResolution('No video track'); return }
    const settings = track.getSettings()
    setResolution(`${settings.width ?? '?'}×${settings.height ?? '?'}`)

    // Listen for track ended
    track.onended = () => { setResolution('No signal'); setStatus('idle') }
  }, [stream])

  // Bitrate estimation from recorder chunks (passed in from recorder)
  const updateBitrate = (bytesPerSecond: number) => {
    const mbps = (bytesPerSecond * 8 / 1000000).toFixed(2)
    setBitrate(`${mbps} Mbps`)
  }

  return { fps, resolution, bitrate, status, setStatus, updateBitrate }
}
```
