```typescript
// src/utils/orderFlowEngine.ts

// DATA SOURCES (all free):
// 1. CFTC Commitment of Traders — updated every Friday
// 2. Open Interest data from free APIs
// 3. Retail sentiment from Myfxbook (free)
// 4. Options market data approximation

const CFTC_API = 'https://publicreporting.cftc.gov/api/views/6dca-aqww/rows.json'
const CFTC_GOLD_FUTURES = 'https://publicreporting.cftc.gov/api/views/dcoq-sdmj/rows.json'

// Map CFTC contract names to our pairs
const CFTC_CONTRACT_MAP: Record<string, string> = {
  'EURO FX': 'EURUSD',
  'BRITISH POUND': 'GBPUSD',
  'JAPANESE YEN': 'USDJPY',
  'SWISS FRANC': 'USDCHF',
  'CANADIAN DOLLAR': 'USDCAD',
  'AUSTRALIAN DOLLAR': 'AUDUSD',
  'NEW ZEALAND DOLLAR': 'NZDUSD',
  'GOLD': 'XAUUSD',
  'SILVER': 'XAGUSD',
  'CRUDE OIL': 'USOIL',
  'E-MINI S&P 500': 'SPX500',
  'NASDAQ-100 MINI': 'NAS100',
}

export interface COTData {
  pair: string
  reportDate: string
  commercialLong: number
  commercialShort: number
  commercialNet: number
  nonCommercialLong: number // Large speculators (hedge funds)
  nonCommercialShort: number
  nonCommercialNet: number
  retailLong: number // Small traders
  retailShort: number
  retailNet: number
  openInterest: number
  weeklyChange: {
    commercialNet: number
    nonCommercialNet: number
    retailNet: number
  }
  // Derived signals
  institutionalBias: 'Strongly Bullish' | 'Bullish' | 'Neutral' | 'Bearish' | 'Strongly Bearish'
  extremePositioning: boolean // When at 52-week high/low
  contraryIndicator: boolean // Retail at extreme = opposite expected
  signal: string
}

export async function fetchCOTData(): Promise<COTData[]> {
  const results: COTData[] = []

  try {
    // Fetch from CFTC public API
    const res = await fetch(
      `${CFTC_API}?$limit=500&$order=report_date_as_yyyy_mm_dd DESC`,
      { headers: { Accept: 'application/json' } }
    )
    const data = await res.json()
    const rows = data.data || []

    // Group by contract and get latest + previous week
    const contractGroups: Record<string, any[]> = {}
    rows.forEach((row: any) => {
      const name = row[3] // contract name
      if (!contractGroups[name]) contractGroups[name] = []
      contractGroups[name].push(row)
    })

    Object.entries(contractGroups).forEach(([contractName, rows]) => {
      const pair = CFTC_CONTRACT_MAP[contractName]
      if (!pair || rows.length < 2) return

      const latest = rows[0]
      const previous = rows[1]

      const nonCommLong = parseInt(latest[9] || '0')
      const nonCommShort = parseInt(latest[10] || '0')
      const nonCommNet = nonCommLong - nonCommShort

      const prevNonCommLong = parseInt(previous[9] || '0')
      const prevNonCommShort = parseInt(previous[10] || '0')
      const prevNonCommNet = prevNonCommLong - prevNonCommShort

      const commLong = parseInt(latest[6] || '0')
      const commShort = parseInt(latest[7] || '0')
      const commNet = commLong - commShort

      const retailLong = parseInt(latest[12] || '0')
      const retailShort = parseInt(latest[13] || '0')
      const retailNet = retailLong - retailShort

      const openInterest = parseInt(latest[5] || '0')

      // Determine institutional bias
      const netChangeWeek = nonCommNet - prevNonCommNet
      let institutionalBias: COTData['institutionalBias'] = 'Neutral'
      if (nonCommNet > 50000) institutionalBias = 'Strongly Bullish'
      else if (nonCommNet > 20000) institutionalBias = 'Bullish'
      else if (nonCommNet < -50000) institutionalBias = 'Strongly Bearish'
      else if (nonCommNet < -20000) institutionalBias = 'Bearish'

      // Extreme positioning check (simplified)
      const extremePositioning = Math.abs(nonCommNet) > 100000

      // Contrary indicator: if retail at extreme, opposite likely
      const retailExtreme = Math.abs(retailNet) > 30000
      const retailBullish = retailNet > 0
      const contraryIndicator = retailExtreme

      results.push({
        pair,
        reportDate: latest[4],
        commercialLong: commLong,
        commercialShort: commShort,
        commercialNet: commNet,
        nonCommercialLong: nonCommLong,
        nonCommercialShort: nonCommShort,
        nonCommercialNet: nonCommNet,
        retailLong,
        retailShort,
        retailNet,
        openInterest,
        weeklyChange: {
          commercialNet: commNet - (parseInt(previous[6] || '0') - parseInt(previous[7] || '0')),
          nonCommercialNet: netChangeWeek,
          retailNet: retailNet - (parseInt(previous[12] || '0') - parseInt(previous[13] || '0'))
        },
        institutionalBias,
        extremePositioning,
        contraryIndicator,
        signal: generateCOTSignal(nonCommNet, netChangeWeek, retailNet, retailExtreme, retailBullish)
      })
    })
  } catch (e) {
    console.error('CFTC fetch failed:', e)
  }

  return results
}

function generateCOTSignal(
  instNet: number,
  instWeeklyChange: number,
  retailNet: number,
  retailExtreme: boolean,
  retailBullish: boolean
): string {
  const signals: string[] = []

  if (instNet > 50000 && instWeeklyChange > 5000)
    signals.push('Institutions heavily long and ADDING — strong bull signal')
  else if (instNet < -50000 && instWeeklyChange < -5000)
    signals.push('Institutions heavily short and ADDING — strong bear signal')
  else if (Math.abs(instWeeklyChange) > 10000)
    signals.push(`Institutions rapidly ${instWeeklyChange > 0 ? 'covering shorts / going long' : 'reducing longs / going short'}`)

  if (retailExtreme && retailBullish && instNet < 0)
    signals.push('Retail at extreme long while institutions short — BEARISH contrarian signal')
  else if (retailExtreme && !retailBullish && instNet > 0)
    signals.push('Retail at extreme short while institutions long — BULLISH contrarian signal')

  return signals.join('. ') || 'No clear COT signal this week'
}

// Myfxbook retail sentiment
export async function fetchRetailSentiment(pair: string): Promise<{ long: number; short: number }> {
  try {
    const symbol = pair.toLowerCase()
    const url = `https://www.myfxbook.com/api/get-community-outlook.json?pair=${symbol}`
    const res = await fetch(`https://api.allorigins.win/get?url=${encodeURIComponent(url)}`)
    const wrapper = await res.json()
    const data = JSON.parse(wrapper.contents)
    const sym = data?.symbols?.[0]
    if (sym) {
      return {
        long: parseFloat(sym.longPercentage),
        short: parseFloat(sym.shortPercentage)
      }
    }
  } catch {}
  return { long: 50, short: 50 }
}
```

**=== FEATURE 2: SMART MONEY DIVERGENCE DETECTOR ===**

**Route:** `/divergence`
**Sidebar:** Under AI TOOLS section

Divergence between price and institutional positioning is one of the most powerful signals in trading. When institutions are accumulating longs while price is falling — that is the real bottom.

Create `src/pages/DivergenceDetector.tsx`:

```typescript
// Types of divergence to detect:

interface DivergenceSignal {
  type: 'price_cot' | 'retail_institutional' | 'dxy_gold' | 'vix_risk' | 'volume_price' | 'momentum_price'
  pair: string
  direction: 'bullish' | 'bearish'
  strength: number // 1-10
  description: string
  historicalAccuracy: number // % of times this signal was correct historically
  priceLevel: number
  targetLevel: number
  timeframe: string
  detectedAt: Date
}

// DIVERGENCE TYPE 1: COT vs Price divergence
// When institutions are net long but price is falling = accumulation phase = BUY signal
// When institutions are net short but price is rising = distribution phase = SELL signal
async function detectCOTPriceDivergence(cotData: COTData, currentPrice: number, priceChange7d: number): Promise<DivergenceSignal | null> {
  const instBullish = cotData.nonCommercialNet > 0
  const priceFalling = priceChange7d < -0.5

  if (instBullish && priceFalling) {
    return {
      type: 'price_cot',
      pair: cotData.pair,
      direction: 'bullish',
      strength: Math.min(10, Math.round(cotData.nonCommercialNet / 20000)),
      description: `Institutions NET LONG ${cotData.nonCommercialNet.toLocaleString()} contracts while price fell ${Math.abs(priceChange7d).toFixed(2)}% this week. This is classic accumulation. Smart money is absorbing retail panic selling.`,
      historicalAccuracy: 71,
      priceLevel: currentPrice,
      targetLevel: currentPrice * 1.02,
      timeframe: 'Weekly',
      detectedAt: new Date()
    }
  }

  const instBearish = cotData.nonCommercialNet < 0
  const priceRising = priceChange7d > 0.5

  if (instBearish && priceRising) {
    return {
      type: 'price_cot',
      pair: cotData.pair,
      direction: 'bearish',
      strength: Math.min(10, Math.round(Math.abs(cotData.nonCommercialNet) / 20000)),
      description: `Institutions NET SHORT ${Math.abs(cotData.nonCommercialNet).toLocaleString()} contracts while price rose ${priceChange7d.toFixed(2)}% this week. Distribution phase. Smart money selling into retail buying.`,
      historicalAccuracy: 68,
      priceLevel: currentPrice,
      targetLevel: currentPrice * 0.98,
      timeframe: 'Weekly',
      detectedAt: new Date()
    }
  }

  return null
}

// DIVERGENCE TYPE 2: DXY vs Gold divergence
// Gold normally moves inverse to DXY
// When both rise together = unusual strength in gold = very bullish for gold
// When both fall together = safe haven selling = very bearish for gold
async function detectDXYGoldDivergence(dxyPrice: number, goldPrice: number, dxyChange: number, goldChange: number): Promise<DivergenceSignal | null> {
  const bothRising = dxyChange > 0.3 && goldChange > 0.3
  const bothFalling = dxyChange < -0.3 && goldChange < -0.3

  if (bothRising) {
    return {
      type: 'dxy_gold',
      pair: 'XAUUSD',
      direction: 'bullish',
      strength: 8,
      description: `RARE: Gold rising (+${goldChange.toFixed(2)}%) alongside DXY (+${dxyChange.toFixed(2)}%). This is abnormal positive divergence — gold has extreme underlying demand overcoming dollar strength. Historically leads to accelerated gold upside.`,
      historicalAccuracy: 74,
      priceLevel: goldPrice,
      targetLevel: goldPrice * 1.015,
      timeframe: 'Daily',
      detectedAt: new Date()
    }
  }

  return null
}

// DIVERGENCE TYPE 3: Retail vs Institutional divergence
// When retail is at extreme (>70% one side) and institutions opposite = contrarian signal
async function detectRetailInstitutionalDivergence(sentiment: { long: number; short: number }, cotData: COTData, currentPrice: number): Promise<DivergenceSignal | null> {
  const retailExtremeLong = sentiment.long > 70
  const retailExtremeShort = sentiment.short > 70
  const instLong = cotData.nonCommercialNet > 20000
  const instShort = cotData.nonCommercialNet < -20000

  if (retailExtremeLong && instShort) {
    return {
      type: 'retail_institutional',
      pair: cotData.pair,
      direction: 'bearish',
      strength: Math.round((sentiment.long - 50) / 5),
      description: `${sentiment.long}% of retail traders are LONG while institutions are NET SHORT ${Math.abs(cotData.nonCommercialNet).toLocaleString()} contracts. When retail crowd is this one-sided, institutions typically engineer a move against them to take their stop losses. BEARISH contrarian signal.`,
      historicalAccuracy: 69,
      priceLevel: currentPrice,
      targetLevel: currentPrice * 0.985,
      timeframe: 'Daily-Weekly',
      detectedAt: new Date()
    }
  }

  if (retailExtremeShort && instLong) {
    return {
      type: 'retail_institutional',
      pair: cotData.pair,
      direction: 'bullish',
      strength: Math.round((sentiment.short - 50) / 5),
      description: `${sentiment.short}% of retail traders are SHORT while institutions are NET LONG ${cotData.nonCommercialNet.toLocaleString()} contracts. Classic setup for a short squeeze. Institutions will push price up to harvest retail stop losses above.`,
      historicalAccuracy: 72,
      priceLevel: currentPrice,
      targetLevel: currentPrice * 1.015,
      timeframe: 'Daily-Weekly',
      detectedAt: new Date()
    }
  }

  return null
}

// Master divergence scanner
export async function scanAllDivergences(prices: Record<string, number>, cotData: COTData[]): Promise<DivergenceSignal[]> {
  const signals: DivergenceSignal[] = []

  for (const cot of cotData) {
    const price = prices[cot.pair] || 0
    if (!price) continue

    // Approximate 7-day price change (use stored historical if available)
    const storedHistory = JSON.parse(localStorage.getItem(`price_history_${cot.pair}`) || '[]')
    const priceChange7d = storedHistory.length > 0
      ? ((price - storedHistory[0]) / storedHistory[0]) * 100
      : 0

    // COT vs Price divergence
    const cotDiv = await detectCOTPriceDivergence(cot, price, priceChange7d)
    if (cotDiv) signals.push(cotDiv)

    // Retail vs Institutional
    const sentiment = await fetchRetailSentiment(cot.pair)
    const retailInstDiv = await detectRetailInstitutionalDivergence(sentiment, cot, price)
    if (retailInstDiv) signals.push(retailInstDiv)
  }

  // DXY vs Gold
  if (prices.DXY && prices.XAUUSD) {
    const dxyHistory = JSON.parse(localStorage.getItem('price_history_DXY') || '[]')
    const goldHistory = JSON.parse(localStorage.getItem('price_history_XAUUSD') || '[]')
    const dxyChange = dxyHistory.length > 0 ? ((prices.DXY - dxyHistory[0]) / dxyHistory[0]) * 100 : 0
    const goldChange = goldHistory.length > 0 ? ((prices.XAUUSD - goldHistory[0]) / goldHistory[0]) * 100 : 0
    const dxyGoldDiv = await detectDXYGoldDivergence(prices.DXY, prices.XAUUSD, dxyChange, goldChange)
    if (dxyGoldDiv) signals.push(dxyGoldDiv)
  }

  return signals.sort((a, b) => b.strength - a.strength)
}
```

```typescript
// src/utils/liquidityEngine.ts

interface LiquidityLevel {
  price: number
  type: 'BSL' | 'SSL' | 'EQH' | 'EQL' | 'PDH' | 'PDL' | 'PWH' | 'PWL' | 'ROUND'
  label: string
  strength: 'High' | 'Medium' | 'Low'
  distanceFromCurrentPrice: number
  distancePips: number
  probability: number // % chance of being targeted next
  reason: string
  timeframe: string
}

// Calculate liquidity levels from price data
export async function calculateLiquidityLevels(pair: string, currentPrice: number): Promise<LiquidityLevel[]> {
  const levels: LiquidityLevel[] = []

  // Round numbers (psychological levels) — major magnets
  const roundInterval = currentPrice > 100 ? 50 : currentPrice > 10 ? 5 : 0.005
  const nearestRounds = []
  for (let i = -5; i <= 5; i++) {
    const roundLevel = Math.round(currentPrice / roundInterval) * roundInterval + (i * roundInterval)
    if (Math.abs(roundLevel - currentPrice) > roundInterval * 0.1) {
      nearestRounds.push(roundLevel)
    }
  }

  nearestRounds.forEach(level => {
    const isAbove = level > currentPrice
    const distance = Math.abs(level - currentPrice)
    levels.push({
      price: level,
      type: 'ROUND',
      label: `Round number ${level.toFixed(pair.includes('JPY') ? 1 : 2)}`,
      strength: distance < roundInterval ? 'High' : 'Medium',
      distanceFromCurrentPrice: level - currentPrice,
      distancePips: Math.abs(level - currentPrice) * (pair.includes('JPY') ? 100 : 10000),
      probability: distance < roundInterval ? 65 : 45,
      reason: 'Round number — retail orders cluster here. Stops placed just beyond.',
      timeframe: 'All timeframes'
    })
  })

  // Previous Day High/Low — always major liquidity targets
  // Approximate PDH/PDL from current price (AI will refine)
  const estimatedPDH = currentPrice * 1.008
  const estimatedPDL = currentPrice * 0.992

  levels.push({
    price: estimatedPDH,
    type: 'PDH',
    label: 'Previous Day High',
    strength: 'High',
    distanceFromCurrentPrice: estimatedPDH - currentPrice,
    distancePips: (estimatedPDH - currentPrice) * 10000,
    probability: 72,
    reason: 'Buy stops cluster above PDH. Smart money will sweep this before reversing.',
    timeframe: 'D1'
  })

  levels.push({
    price: estimatedPDL,
    type: 'PDL',
    label: 'Previous Day Low',
    strength: 'High',
    distanceFromCurrentPrice: estimatedPDL - currentPrice,
    distancePips: (currentPrice - estimatedPDL) * 10000,
    probability: 70,
    reason: 'Sell stops cluster below PDL. Smart money will grab these before reversing.',
    timeframe: 'D1'
  })

  // Get AI to identify the specific SMC liquidity levels
  const aiLevels = await callAlchemistAI(`
Identify all liquidity levels for ${pair} at current price ${currentPrice}.

Return ONLY this JSON array:
[
  {
    "price": ${(currentPrice * 1.005).toFixed(2)},
    "type": "EQH",
    "label": "Equal highs at X",
    "strength": "High",
    "probability": 78,
    "reason": "Three equal highs at this level — buy stops stacked here",
    "timeframe": "H1"
  }
]

Include:
- Equal highs (EQH) — buy stops above
- Equal lows (EQL) — sell stops below
- Previous week high/low (PWH/PWL)
- Previous day high/low (PDH/PDL)
- Any obvious stop clusters based on swing structure
- Round numbers near current price

Use actual price ${currentPrice} as reference. Return 8-12 levels.
  `, currentPrice, pair)

  try {
    const jsonMatch = aiLevels.match(/\[[\s\S]*\]/)
    if (jsonMatch) {
      const parsed: LiquidityLevel[] = JSON.parse(jsonMatch[0])
      parsed.forEach(l => {
        l.distanceFromCurrentPrice = l.price - currentPrice
        l.distancePips = Math.abs(l.price - currentPrice) * (pair.includes('JPY') ? 100 : 10000)
      })
      levels.push(...parsed)
    }
  } catch {}

  // Sort by distance from current price
  return levels.sort((a, b) =>
    Math.abs(a.distanceFromCurrentPrice) - Math.abs(b.distanceFromCurrentPrice)
  )
}

// Determine which liquidity will be targeted first
export function getNextLiquidityTarget(levels: LiquidityLevel[], bias: 'Bullish' | 'Bearish'): LiquidityLevel | null {
  if (bias === 'Bullish') {
    return levels
      .filter(l => l.price > 0 && l.distanceFromCurrentPrice > 0)
      .sort((a, b) => b.probability - a.probability)[0] || null
  } else {
    return levels
      .filter(l => l.distanceFromCurrentPrice < 0)
      .sort((a, b) => b.probability - a.probability)[0] || null
  }
}
```

```typescript
    const check = () => {
      alerts.filter(a => a.active && !a.triggered).forEach(alert => {
        const currentPrice = prices[alert.pair]
        if (!currentPrice) return

        let shouldTrigger = false

        switch(alert.trigger) {
          case 'price_above':
            shouldTrigger = currentPrice >= Number(alert.triggerValue)
            break
          case 'price_below':
            shouldTrigger = currentPrice <= Number(alert.triggerValue)
            break
          case 'price_touch':
            const tolerance = currentPrice * 0.0002
            shouldTrigger = Math.abs(currentPrice - Number(alert.triggerValue)) <= tolerance
            break
          case 'ob_touch':
            const [obLow, obHigh] = String(alert.triggerValue).split('-').map(Number)
            shouldTrigger = currentPrice >= obLow && currentPrice <= obHigh
            break
          case 'session_open':
            const utcHour = new Date().getUTCHours()
            const sessionHour = Number(alert.triggerValue)
            shouldTrigger = utcHour === sessionHour && new Date().getUTCMinutes() < 5
            break
        }

        if (shouldTrigger) {
          triggerAlert(alert, currentPrice)
        }
      })
    }

    const interval = setInterval(check, 5000)
    return () => clearInterval(interval)
  }, [prices, alerts])
}

function triggerAlert(alert: Alert, currentPrice: number) {
  // Sound
  if (alert.sound) {
    streamAlertService.playSound('price_alert')
  }

  // TTS
  if (alert.tts) {
    const msg = `Alert: ${alert.pair} has ${alert.trigger.replace('_', ' ')} ${alert.triggerValue}. Current price: ${currentPrice.toFixed(2)}`
    streamAlertService.speakAlert(msg)
  }

  // Browser notification
  if (alert.notification && Notification.permission === 'granted') {
    new Notification(`🚨 ${alert.pair} Alert: ${alert.label}`, {
      body: `Price: ${currentPrice.toFixed(2)} | ${alert.notes}`,
      icon: '/logo.png',
      badge: '/logo.png'
    })
  }

  // Telegram
  if (alert.telegram) {
    const botToken = localStorage.getItem('jjnexus_telegram_token') || ''
    const targets = JSON.parse(localStorage.getItem('jjnexus_telegram_targets') || '[]')
    if (botToken && targets.length > 0) {
      targets.forEach((target: any) => {
        fetch(`https://api.telegram.org/bot${botToken}/sendMessage`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            chat_id: target.chatId,
            text: `🚨 <b>${alert.pair} ALERT TRIGGERED</b>\n\n${alert.label}\nPrice: $${currentPrice.toFixed(2)}\n${alert.notes}\n\n<i>— JJ NEXUS PRO Alert System</i>`,
            parse_mode: 'HTML'
          })
        })
      })
    }
  }
}
```
