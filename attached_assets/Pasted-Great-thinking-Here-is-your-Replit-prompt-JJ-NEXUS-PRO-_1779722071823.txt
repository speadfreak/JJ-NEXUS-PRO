async function startCamera(facing = 'environment') {
  try {
    if (stream) stream.getTracks().forEach(t => t.stop())
    stream = await navigator.mediaDevices.getUserMedia({
      video: {
        facingMode: { ideal: facing },
        width: { ideal: 1920, max: 3840 },
        height: { ideal: 1080, max: 2160 },
        frameRate: { ideal: 60, max: 60 }
      },
      audio: { echoCancellation: true, noiseSuppression: true, sampleRate: 44100 }
    })
    document.getElementById('preview').srcObject = stream
    currentFacing = facing

    // Show quality info
    const track = stream.getVideoTracks()[0]
    const settings = track.getSettings()
    document.getElementById('quality-badge').style.display = 'block'
    document.getElementById('res-display').textContent = `${settings.width}×${settings.height}`
    document.getElementById('fps-display').textContent = `${settings.frameRate} fps`

    // Measure real FPS
    startFPSMeasure()

    // Re-send stream to relay if already connected
    if (isConnected && ws) sendStreamToRelay()

    return true
  } catch (e) {
    setStatus('error', 'CAMERA ERROR')
    return false
  }
}

async function connectToStudio() {
  if (!stream) {
    const ok = await startCamera(currentFacing)
    if (!ok) return
  }

  setStatus('connecting', 'CONNECTING...')

  try {
    ws = new WebSocket(RELAY_WS_URL)

    ws.onopen = () => {
      isConnected = true
      setStatus('connected', 'LIVE TO STUDIO')
      document.getElementById('btn-connect').classList.add('connected')
      document.getElementById('btn-connect').textContent = '⏹'

      // Send camera info to studio
      const track = stream.getVideoTracks()[0]
      const settings = track.getSettings()
      ws.send(JSON.stringify({
        type: 'PHONE_CAMERA_INIT',
        width: settings.width,
        height: settings.height,
        fps: settings.frameRate,
        facing: currentFacing,
        deviceId: track.getSettings().deviceId,
        label: track.label
      }))

      // Start keep-alive ping every 15 seconds
      keepAliveTimer = setInterval(() => {
        if (ws?.readyState === WebSocket.OPEN) {
          ws.send(JSON.stringify({ type: 'PING', timestamp: Date.now() }))
        }
      }, 15000)

      // Notify parent window if opened from studio
      postToStudio({ type: 'PHONE_CAMERA_READY', width: settings.width, height: settings.height })

      // Store connection token in localStorage for auto-reconnect
      localStorage.setItem('jjnexus_cam_session', JSON.stringify({
        connected: true,
        timestamp: Date.now(),
        relay: RELAY_WS_URL
      }))

      sendStreamToRelay()
    }

    ws.onmessage = (e) => {
      try {
        const msg = JSON.parse(e.data)
        if (msg.type === 'PONG') {} // keepalive confirmed
        if (msg.type === 'STUDIO_COMMAND') handleStudioCommand(msg)
      } catch {}
    }

    ws.onclose = (e) => {
      if (isConnected) {
        isConnected = false
        setStatus('connecting', 'RECONNECTING...')
        startAutoReconnect()
      }
    }

    ws.onerror = () => {
      setStatus('error', 'CONNECTION FAILED')
      startAutoReconnect()
    }

  } catch (e) {
    setStatus('error', 'FAILED')
  }
}

function handleStudioCommand(msg) {
  switch(msg.command) {
    case 'flip': flipCamera(); break
    case 'mirror': toggleMirror(); break
    case 'torch': toggleTorch(); break
    case 'brightness': document.getElementById('brightness').value = msg.value; applyFilters(); break
    case 'zoom': document.getElementById('zoom').value = msg.value * 10; applyZoom(); break
    case 'snapshot': takeSnapshot(); break
  }
}

function disconnectFromStudio() {
  isConnected = false
  if (keepAliveTimer) clearInterval(keepAliveTimer)
  if (reconnectTimer) clearTimeout(reconnectTimer)

function sendStreamToRelay() {
  if (!stream || !ws || ws.readyState !== WebSocket.OPEN) return

  // Send stream metadata — studio uses this to create a video element
  // The actual video frames are sent via WebRTC peer connection
  ws.send(JSON.stringify({
    type: 'CAMERA_STREAM_READY',
    streamId: stream.id,
    tracks: stream.getTracks().map(t => ({
      kind: t.kind,
      label: t.label,
      settings: t.getSettings()
    }))
  }))
}

function startFPSMeasure() {
  if (fpsTimer) cancelAnimationFrame(fpsTimer)
  let lastTime = performance.now()
  let frames = 0
  const measure = () => {
    frames++
    const now = performance.now()
    if (now - lastTime >= 1000) {

      document.getElementById('kb-display').textContent = `${Math.round(frames * 40)} kbps est`
      frames = 0
      lastTime = now
    }
    fpsTimer = requestAnimationFrame(measure)
  }
  fpsTimer = requestAnimationFrame(measure)
}

    window.addEventListener('message', handleMessage)

    // Poll localStorage for phone camera events
    const poller = setInterval(() => {
      const event = localStorage.getItem('jjnexus_cam_event')
      if (event) {
        const data = JSON.parse(event)
        if (Date.now() - data.ts < 5000) {
          if (data.type === 'PHONE_CAMERA_READY') handlePhoneConnected(data)
          if (data.type === 'PING') {
            setLastSeen(new Date())
            setSignalStrength(4)
          }
        }
        localStorage.removeItem('jjnexus_cam_event')
      }
    }, 1000)

    return () => {
      window.removeEventListener('message', handleMessage)
      clearInterval(poller)
    }
  }, [])

  const handlePhoneConnected = (data: any) => {
    const info: PhoneInfo = {
      width: data.width || 1920,
      height: data.height || 1080,
      fps: data.fps || 60,
      facing: data.facing || 'environment',
      label: data.label || 'Phone Camera'
    }
    setPhoneInfo(info)
    setConnectionState('connected')
    setLastSeen(new Date())
    setSignalStrength(4)

    // Save session
    localStorage.setItem('jjnexus_phone_session', JSON.stringify({
      timestamp: Date.now(),
      phoneInfo: info
    }))
    setSessionSaved(true)
  }

  // Send command to phone camera
  const sendCommand = (command: string, value?: any) => {
    localStorage.setItem('jjnexus_cam_event', JSON.stringify({
      type: 'STUDIO_COMMAND',
      command,
      value,
      ts: Date.now()
    }))
  }

  const openPhonePage = () => {
    window.open(phoneUrl, 'JJNexusCam',
      'width=420,height=800,scrollbars=no,toolbar=no,menubar=no,status=no')
    setConnectionState('connecting')

    // Timeout if not connected in 60 seconds
    setTimeout(() => {
      if (connectionState === 'connecting') {
        setConnectionState('disconnected')
      }
    }, 60000)
  }

  const disconnect = () => {
    setConnectionState('disconnected')
    setPhoneInfo(null)
    setSessionSaved(false)
    localStorage.removeItem('jjnexus_phone_session')
  }

  const reconnectQuickly = () => {
    // Just reopen the phone page — it will auto-reconnect
    openPhonePage()
  }

  const copyUrl = () => navigator.clipboard.writeText(phoneUrl)

  const stateColor = {
    disconnected: '#666',
    connecting: '#eab308',
    connected: '#16a34a',
    reconnecting: '#eab308'
  }[connectionState]

  const stateLabel = {
    disconnected: 'Not Connected',
    connecting: 'Waiting for phone...',
    connected: `Connected ${phoneInfo ? `— ${phoneInfo.width}×${phoneInfo.height} @ ${phoneInfo.fps}fps` : ''}`,
    reconnecting: 'Reconnecting...'
  }[connectionState]

  return (
    <div style={{
      background: 'rgba(212,175,55,0.03)',
      border: `1px solid ${connectionState === 'connected' ? 'rgba(22,163,74,0.4)' : 'rgba(212,175,55,0.15)'}`,
      borderRadius: 12, padding: 16, transition: 'border-color 0.3s'
    }}>
      {/* Header */}
      <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: 12 }}>
        <div>
          <div style={{ fontSize: 13, fontWeight: 700, color: '#D4AF37' }}>📱 Phone HD Camera</div>
          <div style={{ fontSize: 11, color: stateColor, marginTop: 2, display: 'flex', alignItems: 'center', gap: 6 }}>
            <div style={{
              width: 7, height: 7, borderRadius: '50%', background: stateColor,
              boxShadow: connectionState === 'connected' ? `0 0 8px ${stateColor}` : 'none',
              animation: connectionState === 'connecting' ? 'pulse 1s infinite' : 'none'
            }}/>
            {stateLabel}
          </div>
        </div>
        {/* Signal strength */}
        {connectionState === 'connected' && (
          <div style={{ display: 'flex', alignItems: 'flex-end', gap: 2 }}>
            {[1,2,3,4].map(i => (
              <div key={i} style={{
                width: 4, height: 4 + i * 4, borderRadius: 1,
                background: i <= signalStrength ? '#16a34a' : '#333'
              }}/>
            ))}
          </div>
        )}
      </div>

      {/* Connected state */}
      {connectionState === 'connected' && phoneInfo && (
        <div style={{ marginBottom: 12 }}>
          <div style={{
            background: 'rgba(22,163,74,0.08)', border: '1px solid rgba(22,163,74,0.2)',
            borderRadius: 8, padding: '10px 14px', marginBottom: 10
          }}>
            <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: 6 }}>
              {[
                ['Resolution', `${phoneInfo.width}×${phoneInfo.height}`],
                ['Frame Rate', `${phoneInfo.fps} fps`],
                ['Camera', phoneInfo.facing === 'environment' ? '📸 Rear' : '🤳 Front'],
                ['Last seen', lastSeen ? `${Math.floor((Date.now() - lastSeen.getTime()) / 1000)}s ago` : '—']
              ].map(([label, value]) => (
                <div key={label}>
                  <div style={{ fontSize: 9, color: '#555', textTransform: 'uppercase', letterSpacing: 1 }}>{label}</div>
                  <div style={{ fontSize: 12, color: '#fff', fontWeight: 600, marginTop: 2 }}>{value}</div>
                </div>
              ))}
            </div>
          </div>

          {/* Remote controls */}
          <div style={{ marginBottom: 8 }}>
            <div style={{ fontSize: 10, color: '#666', marginBottom: 6, textTransform: 'uppercase', letterSpacing: 1 }}>
              Remote Control Phone Camera
            </div>
            <div style={{ display: 'flex', gap: 6, flexWrap: 'wrap' }}>
              {[
                { label: '🔄 Flip', cmd: 'flip' },
                { label: '↔️ Mirror', cmd: 'mirror' },
                { label: '🔦 Torch', cmd: 'torch' },
                { label: '📸 Snap', cmd: 'snapshot' },
              ].map(({ label, cmd }) => (
                <button key={cmd} onClick={() => sendCommand(cmd)}
                  style={{
                    padding: '6px 12px', background: 'rgba(212,175,55,0.1)',
                    border: '1px solid rgba(212,175,55,0.3)', borderRadius: 6,
                    color: '#D4AF37', fontSize: 11, cursor: 'pointer', fontWeight: 600
                  }}>
                  {label}
                </button>
              ))}
            </div>
          </div>

          <button onClick={disconnect} style={{
            width: '100%', padding: '8px', background: 'transparent',
            border: '1px solid rgba(220,38,38,0.3)', borderRadius: 8,
            color: '#dc2626', fontSize: 11, cursor: 'pointer'
          }}>
            Disconnect Phone Camera
          </button>
        </div>
      )}

      {/* Quick reconnect when session saved but disconnected */}
      {connectionState === 'disconnected' && sessionSaved && phoneInfo && (
        <div style={{
          background: 'rgba(212,175,55,0.08)', border: '1px solid rgba(212,175,55,0.2)',
          borderRadius: 8, padding: 12, marginBottom: 12
        }}>
          <div style={{ fontSize: 12, color: '#D4AF37', fontWeight: 700, marginBottom: 4 }}>
            ⚡ Last session found
          </div>
          <div style={{ fontSize: 11, color: '#888', marginBottom: 10 }}>
            {phoneInfo.width}×{phoneInfo.height} @ {phoneInfo.fps}fps — tap to reconnect instantly
          </div>
          <button onClick={reconnectQuickly} style={{
            width: '100%', padding: '10px', background: '#D4AF37',
            border: 'none', borderRadius: 8, color: '#000',
            fontWeight: 700, fontSize: 13, cursor: 'pointer'
          }}>
            ⚡ Reconnect Instantly
          </button>
        </div>
      )}

      {/* First time connection */}
      {connectionState === 'disconnected' && !sessionSaved && (
        <div>
          {/* QR Code */}
          {qrDataUrl && (
            <div style={{ textAlign: 'center', marginBottom: 12 }}>
              <div style={{ fontSize: 10, color: '#666', marginBottom: 8, textTransform: 'uppercase', letterSpacing: 1 }}>
                Scan with your phone camera
              </div>
              <img src={qrDataUrl} alt="QR code to open phone camera page"
                style={{ width: 140, height: 140, border: '2px solid rgba(212,175,55,0.3)', borderRadius: 8 }}/>
              <div style={{ fontSize: 9, color: '#555', marginTop: 6 }}>
                Opens the camera app on your phone
              </div>
            </div>
          )}

          <button onClick={openPhonePage} style={{
            width: '100%', padding: '12px', background: '#D4AF37',
            border: 'none', borderRadius: 8, color: '#000',
            fontWeight: 700, fontSize: 13, cursor: 'pointer', marginBottom: 8
          }}>
            📱 Open Phone Camera
          </button>

          <button onClick={copyUrl} style={{
            width: '100%', padding: '8px', background: 'transparent',
            border: '1px solid rgba(212,175,55,0.3)', borderRadius: 8,
            color: '#D4AF37', fontSize: 11, cursor: 'pointer', marginBottom: 10
          }}>
            Copy Link — Open on Phone Browser
          </button>

          <div style={{
            background: 'rgba(212,175,55,0.05)', borderRadius: 8,
            padding: '10px 12px', fontSize: 10, color: '#666', lineHeight: 1.7
          }}>
            💡 Tip: After first connection, install the camera page as an app on your phone home screen for instant reconnect every time.
          </div>
        </div>
      )}

      {connectionState === 'connecting' && (
        <div style={{ textAlign: 'center', padding: '20px 0' }}>
          <div style={{ fontSize: 24, marginBottom: 10 }}>⏳</div>
          <div style={{ fontSize: 12, color: '#eab308' }}>Waiting for phone to connect...</div>
          <div style={{ fontSize: 10, color: '#555', marginTop: 6 }}>Open the link on your phone and tap Connect</div>
        </div>
      )}
    </div>
  )
}

const RemoteCameraControls = () => {
