function resolveFFmpeg(): string {

export function attachStreamRelay(httpServer: HTTPServer) {

function handleVideoChunk(session: Session, data: Buffer) {
  session.lastChunkTime = Date.now()
  session.segmentCount++

  // Write segment to disk
  const segFile = path.join(session.workDir, `seg_${String(session.segmentCount).padStart(6, '0')}.webm`)
  fs.writeFileSync(segFile, data)

  console.log(`[${session.id}] Segment #${session.segmentCount} — ${data.length} bytes`)

  // Validate first chunk is valid WebM (starts with EBML header 0x1A45DFA3)
  if (session.segmentCount === 1) {
    const magic = data.subarray(0, 4).toString('hex')
    console.log(`[${session.id}] First chunk magic: ${magic} (should be 1a45dfa3)`)
    if (!magic.startsWith('1a45')) {
      console.error(`[${session.id}] ❌ Invalid WebM header — got ${magic}`)
      session.ws.send(JSON.stringify({
        type: 'error',
        message: `Invalid video data. Expected WebM header but got: ${magic}. Try a different browser or codec.`
      }))
      cleanupSession(session)
      return
    }
    console.log(`[${session.id}] ✅ Valid WebM header confirmed`)
  }

  // Start FFmpeg after receiving first 2 segments
  if (session.segmentCount === 2 && !session.ffmpeg) {
    session.ws.send(JSON.stringify({
      type: 'server_status',
      message: 'Starting FFmpeg encoder…'
    }))
    spawnFFmpeg(session)
  }

  // For ongoing segments — append to FFmpeg stdin directly
  if (session.ffmpeg?.stdin && !session.ffmpeg.stdin.destroyed && session.segmentCount > 2) {
    try {
      session.ffmpeg.stdin.write(data)
    } catch (e) {
      console.warn(`[${session.id}] stdin write error:`, e)
    }
  }

  // Watchdog reset
  if (session.watchdog) clearInterval(session.watchdog)
  session.watchdog = setInterval(() => {
    if (Date.now() - session.lastChunkTime > 25000) {
      console.log(`[${session.id}] Watchdog — no chunks for 25s`)
      cleanupSession(session)
    }
  }, 10000)
}

    '-sc_threshold', '0',

    '-b:a', audioBitrate,
    '-ar', '44100',

    '-af', 'aresample=async=1:min_hard_comp=0.100000:first_pts=0',

    '-flvflags', 'no_duration_filesize',
    '-reconnect', '1',
    '-reconnect_at_eof', '1',
    '-reconnect_streamed', '1',
    '-reconnect_delay_max', '5',
    session.rtmpUrl

  ffmpeg.stderr?.on('data', (d: Buffer) => {
    const line = d.toString()
    session.stderrLog.push(line)
    if (session.stderrLog.length > 200) session.stderrLog.shift()

    // Log everything for first 8 seconds
    if (Date.now() - session.startTime < 8000) {
      console.log(`[${session.id}] FFmpeg: ${line.trim()}`)
    }

    if (line.includes('frame=') && firstProgress) {
      firstProgress = false

      try {

      const speedMatch = line.match(/speed=\s*([\d.]+x)/)

          fps: fpsMatch?.[1] || '0',
          bitrate: bitrateMatch?.[1] || '0',
          speed: speedMatch?.[1] || '0x'
        }))
      } catch {}
    }

    // Detect RTMP auth failure
    if (line.includes('Connection refused') || line.includes('403') || line.includes('rtmp') && line.includes('error')) {
      console.error(`[${session.id}] RTMP error:`, line.trim())
    }
  })

  ffmpeg.on('close', (code: number) => {

  })

export class BrowserStreamEngine {

  async startStream(config: StreamConfig): Promise<void> {
    if (this.stats.status === 'connecting' || this.stats.status === 'live') return
    this.config = config
    this.emit({ status: 'connecting', error: null, duration: 0 })

    // Guard: validate stream key

      return

    try {
      this.displayStream = await navigator.mediaDevices.getDisplayMedia({

          height: { ideal: 720, max: 1080 },
          frameRate: { ideal: config.fps, max: 60 },
          cursor: 'always'
        } as MediaTrackConstraints,
        audio: false // Never request system audio — causes Bluetooth disconnect

      this.displayStream.getVideoTracks()[0].addEventListener('ended', () => {
        console.log('[BrowserStreamEngine] Screen share ended by user')
        this.stopStream()
      })
    } catch (e: any) {
      const msg = e.name === 'NotAllowedError'
        ? '❌ Screen share permission denied. Click "Share" in the browser dialog, or open this page in a full tab.'
        : `❌ Screen capture failed: ${e.message}`
      this.emit({ status: 'error', error: msg })
      return
    }

    // Microphone
    try {
      const audioConstraints = config.audioDeviceId
        ? {
            deviceId: { exact: config.audioDeviceId },
            echoCancellation: !config.audioDeviceIsBluetooth,
            noiseSuppression: !config.audioDeviceIsBluetooth,
            autoGainControl: !config.audioDeviceIsBluetooth,
            sampleRate: config.audioDeviceIsBluetooth ? 16000 : 48000
          }
        : { echoCancellation: true, noiseSuppression: true, autoGainControl: true }

      this.micStream = await navigator.mediaDevices.getUserMedia({ audio: audioConstraints })
    } catch {
      this.micStream = null
      console.warn('[BrowserStreamEngine] Mic unavailable — streaming without audio')

    // Combine video + audio
    const tracks: MediaStreamTrack[] = [...this.displayStream.getVideoTracks()]
    if (this.micStream) {
      try {
        const ctx = new AudioContext({ sampleRate: 44100 })
        const dest = ctx.createMediaStreamDestination()
        const compressor = ctx.createDynamicsCompressor()
        ctx.createMediaStreamSource(this.micStream).connect(compressor)
        compressor.connect(dest)
        tracks.push(...dest.stream.getAudioTracks())

        tracks.push(...this.micStream.getAudioTracks())
      }

    const combined = new MediaStream(tracks)

    // Connect WebSocket
    try {
      await this.connectWebSocket()
    } catch (e: any) {
      this.emit({ status: 'error', error: `❌ WebSocket failed: ${e.message}` })

      rtmpUrl: config.rtmpUrl,
      streamKey: config.streamKey,
      videoBitrate: config.videoBitrate,
      audioBitrate: config.audioBitrate,
      fps: config.fps
    }))

    // Start MediaRecorder

  private connectWebSocket(): Promise<void> {
    return new Promise((resolve, reject) => {
      const proto = location.protocol === 'https:' ? 'wss:' : 'ws:'
      const url = `${proto}//${location.host}/api/stream/ws`
      console.log(`[BrowserStreamEngine] Connecting WS: ${url}`)

      const ws = new WebSocket(url)
      const timeout = setTimeout(() => {
        ws.close()
        reject(new Error(`WebSocket timeout — server not responding at ${url}`))
      }, 10000)

      ws.onopen = () => console.log('[BrowserStreamEngine] WS open')

      ws.onmessage = (e) => {
        try {
          const msg = JSON.parse(e.data)
          console.log('[BrowserStreamEngine] Server:', msg.type, msg.message || '')

          if (msg.type === 'connected') {
            clearTimeout(timeout)
            this.ws = ws
            resolve()
          }
          if (msg.type === 'server_status') {
            this.emit({ error: null }) // clear any previous error
            console.log('[BrowserStreamEngine] Server buffering:', msg.message)
          }
          if (msg.type === 'stream_starting') {
            this.emit({ status: 'live', error: null })
            this.startDurationTimer()
          }
          if (msg.type === 'ffmpeg_progress') {
            this.emit({ fps: msg.fps, bitrate: msg.bitrate, speed: msg.speed })
          }
          if (msg.type === 'error') {
            console.error('[BrowserStreamEngine] Server error:', msg.message)
            this.emit({ status: 'error', error: msg.message, debugLog: msg.debugLog })
            this.cleanup()
          }
        } catch {}
      }

      ws.onerror = (e) => {
        clearTimeout(timeout)
        console.error('[BrowserStreamEngine] WS error:', e)
        reject(new Error('WebSocket connection failed'))
      }

      ws.onclose = (e) => {
        if (this.stats.status === 'live') {
          this.emit({ status: 'error', error: `Connection lost (${e.code}). Refresh and try again.` })
          this.cleanup()
        }
      }
    })

  private startMediaRecorder(stream: MediaStream, config: StreamConfig) {
    const mimeTypes = [
      'video/webm;codecs=vp8,opus',
      'video/webm;codecs=vp9,opus',
      'video/webm',
    ]
    const mime = mimeTypes.find(m => MediaRecorder.isTypeSupported(m)) || 'video/webm'
    console.log(`[BrowserStreamEngine] MediaRecorder codec: ${mime}`)

    this.mediaRecorder = new MediaRecorder(stream, {
      mimeType: mime,
      videoBitsPerSecond: parseInt(config.videoBitrate) * 1000,
      audioBitsPerSecond: parseInt(config.audioBitrate) * 1000
    })

    this.mediaRecorder.ondataavailable = (e) => {
      if (e.data.size > 100 && this.ws?.readyState === WebSocket.OPEN) {
        e.data.arrayBuffer().then(buf => {
          try { this.ws?.send(buf) } catch {}
        })
      }
    }

    this.mediaRecorder.onerror = (e: any) => {
      console.error('[BrowserStreamEngine] MediaRecorder error:', e)
      this.emit({ status: 'error', error: `Recording error: ${e.error?.message || 'unknown'}` })
    }

    // Start with 1 second timeslice
    this.mediaRecorder.start(1000)
    console.log('[BrowserStreamEngine] ✅ MediaRecorder started (1s segments)')
  }

  private startDurationTimer() {
    if (this.durationTimer) clearInterval(this.durationTimer)
    this.durationTimer = setInterval(() => {
      this.emit({ duration: this.stats.duration + 1 })
    }, 1000)
  }

  stopStream() {
    this.cleanup()
    this.emit({ status: 'stopped', error: null })
  }

  private cleanup() {
    if (this.durationTimer) { clearInterval(this.durationTimer); this.durationTimer = null }
    if (this.mediaRecorder?.state !== 'inactive') { try { this.mediaRecorder?.stop() } catch {} }
    this.displayStream?.getTracks().forEach(t => t.stop())
    this.micStream?.getTracks().forEach(t => t.stop())
    if (this.ws?.readyState === WebSocket.OPEN) { try { this.ws.send(JSON.stringify({ type: 'stop' })) } catch {} }
    this.ws?.close()
    this.ws = null
    this.mediaRecorder = null
    this.displayStream = null
    this.micStream = null
  }

  static formatDuration(s: number): string {
    const h = Math.floor(s / 3600)
    const m = Math.floor((s % 3600) / 60)
    const sec = s % 60
    return [h, m, sec].map(v => String(v).padStart(2, '0')).join(':')
  }
}

export const streamEngine = new BrowserStreamEngine()
