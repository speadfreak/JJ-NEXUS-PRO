// Send to all targets simultaneously
const sendToAllTargets = async (
  botToken: string,
  targets: TelegramTarget[],
  message: string,
  screenshot: Blob | null,
  pinMessage: boolean
) => {
  const baseUrl = `https://api.telegram.org/bot${botToken}`
  const selectedTargets = targets.filter(t => t.verified)

  const results = await Promise.allSettled(
    selectedTargets.map(async target => {
      let messageId: number | null = null

      if (screenshot) {
        // Send photo with caption
        const form = new FormData()
        form.append('chat_id', target.chatId)
        form.append('photo', screenshot, `${Date.now()}.png`)
        // Telegram caption limit is 1024 chars
        form.append('caption', message.slice(0, 1024))
        form.append('parse_mode', 'HTML')
        const photoRes = await fetch(`${baseUrl}/sendPhoto`, {
          method: 'POST', body: form
        })
        const photoData = await photoRes.json()
        messageId = photoData?.result?.message_id

        // Send rest of message if longer than 1024
        if (message.length > 1024) {
          const contRes = await fetch(`${baseUrl}/sendMessage`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
              chat_id: target.chatId,
              text: message.slice(1024),
              parse_mode: 'HTML'
            })
          })
          const contData = await contRes.json()
          if (!messageId) messageId = contData?.result?.message_id
        }
      } else {
        // Text only
        const textRes = await fetch(`${baseUrl}/sendMessage`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            chat_id: target.chatId,
            text: message,
            parse_mode: 'HTML'
          })
        })
        const textData = await textRes.json()
        messageId = textData?.result?.message_id
      }

      // Pin the message if requested
      if (pinMessage && messageId) {
        await fetch(`${baseUrl}/pinChatMessage`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            chat_id: target.chatId,
            message_id: messageId,
            disable_notification: false
          })
        })
      }

      return { target: target.name, messageId, success: true }
    })
  )

  return results
}

// Verify bot token and get bot info
const verifyBotToken = async (token: string): Promise<{ name: string; username: string } | null> => {
  try {
    const res = await fetch(`https://api.telegram.org/bot${token}/getMe`)
    const data = await res.json()
    if (data.ok) {
      return { name: data.result.first_name, username: data.result.username }
    }
    return null
  } catch { return null }
}

// Verify chat ID (channel or group)
const verifyChatId = async (token: string, chatId: string): Promise<{ title: string; type: string } | null> => {
  try {
    const res = await fetch(`https://api.telegram.org/bot${token}/getChat`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ chat_id: chatId })
    })
    const data = await res.json()
    if (data.ok) {
      return { title: data.result.title || data.result.username, type: data.result.type }
    }
    return null
  } catch { return null }
}

`

  const response = await callAlchemistAI(prompt, livePrice, pair)

// Scan all pairs and return sorted results
export async function scanAllPairs(
  prices: Record<string, number>,
  onProgress: (completed: number, total: number, currentPair: string) => void
): Promise<SetupScore[]> {
  const results: SetupScore[] = []
  const total = SCAN_PAIRS.length

  // Scan in batches of 3 to avoid rate limiting
  for (let i = 0; i < SCAN_PAIRS.length; i += 3) {
    const batch = SCAN_PAIRS.slice(i, i + 3)

    const batchResults = await Promise.allSettled(
      batch.map(async pair => {
        const price = prices[pair] || 0
        if (price === 0) {
          onProgress(i + batch.indexOf(pair) + 1, total, pair)
          return null
        }
        onProgress(i + batch.indexOf(pair) + 1, total, pair)
        return await scorePair(pair, price)
      })
    )

    batchResults.forEach(result => {
      if (result.status === 'fulfilled' && result.value) {
        results.push(result.value)
      }
    })

    // Small delay between batches to respect rate limits
    if (i + 3 < SCAN_PAIRS.length) {
      await new Promise(resolve => setTimeout(resolve, 2000))
    }
  }

  // Sort by score descending
  return results.sort((a, b) => b.score - a.score)
}

      setProgress({ completed, total, currentPair })

    setSetupScores(results)
    setLastScanTime(new Date())
    setIsScanning(false)

    // Auto-send A+ setups to Telegram if enabled
    const aplusSetups = results.filter(r => r.grade === 'A+')
    if (autoSendEnabled && aplusSetups.length > 0 && telegramConfigured) {
      for (const setup of aplusSetups) {
        if (!recentlySent.includes(setup.pair)) {
          await autoSendSetup(setup)
          setRecentlySent(prev => [...prev, setup.pair])
          // Clear from recently sent after 4 hours
          setTimeout(() => {
            setRecentlySent(prev => prev.filter(p => p !== setup.pair))
          }, 4 * 60 * 60 * 1000)
        }
      }
    }

    // Sound alert for A+ setups
    if (aplusSetups.length > 0) {
      playAlertSound('setup_found')
      if (Notification.permission === 'granted') {
        new Notification('🎯 A+ Setup Found!', {
          body: `${aplusSetups[0].pair} — ${aplusSetups[0].bias} — Score: ${aplusSetups[0].score}/100`,
          icon: '/logo.png'
        })
      }
    }
  }

  scan() // immediate first scan

  const interval = setInterval(scan, scanInterval * 60 * 1000) // auto-scan every N minutes
  return () => clearInterval(interval)

class StreamAlertService {
  private audioCache: Map<string, HTMLAudioElement> = new Map()
  private alertQueue: StreamAlert[] = []
  private isProcessing = false
  private overlayCallback: ((alert: StreamAlert, config: AlertConfig) => void) | null = null
  private configs: AlertConfig[] = this.loadConfigs()

  loadConfigs(): AlertConfig[] {
    try {
      const saved = localStorage.getItem('jjnexus_alert_configs')
      return saved ? JSON.parse(saved) : DEFAULT_ALERT_CONFIGS
    } catch { return DEFAULT_ALERT_CONFIGS }
  }

  saveConfigs(configs: AlertConfig[]) {
    this.configs = configs
    localStorage.setItem('jjnexus_alert_configs', JSON.stringify(configs))
  }

  setOverlayCallback(cb: (alert: StreamAlert, config: AlertConfig) => void) {
    this.overlayCallback = cb
  }

  // Preload all alert sounds
  async preloadSounds() {
    for (const config of this.configs) {
      if (config.enabled && config.soundUrl && !this.audioCache.has(config.type)) {
        try {
          const audio = new Audio(config.soundUrl)
          audio.volume = config.volume
          audio.preload = 'auto'
          await audio.load()
          this.audioCache.set(config.type, audio)
        } catch {}
      }
    }
  }

  // Play a specific alert sound
  async playSound(type: AlertType, customUrl?: string) {
    const config = this.configs.find(c => c.type === type)
    if (!config?.enabled) return

    try {
      const url = customUrl || config.soundUrl
      let audio = this.audioCache.get(type)

      if (!audio || audio.src !== url) {
        audio = new Audio(url)
        this.audioCache.set(type, audio)
      }

      audio.volume = config.volume
      audio.currentTime = 0
      await audio.play()
    } catch (e) {
      console.warn('Alert sound failed:', e)
      // Fallback: generate a tone with Web Audio API
      this.generateAlertTone(type)
    }
  }

  // Fallback tone generator when CDN sounds fail
  generateAlertTone(type: AlertType) {
    try {
      const ctx = new AudioContext()
      const tones: Record<AlertType, number[]> = {
        follow: [440, 550, 660],
        subscribe: [550, 660, 770, 880],
        donate: [330, 440, 550, 660, 770],
        gift: [440, 660, 880],
        raid: [220, 330, 440, 550],
        host: [440, 550],
        comment: [440],
        share: [440, 550],
        setup_found: [880, 660, 880],
        price_alert: [440, 880, 440],
        custom: [440, 550, 660]
      }

      const frequencies = tones[type] || [440]
      frequencies.forEach((freq, i) => {
        const osc = ctx.createOscillator()
        const gain = ctx.createGain()
        osc.connect(gain)
        gain.connect(ctx.destination)
        osc.frequency.value = freq
        osc.type = 'sine'
        gain.gain.setValueAtTime(0.3, ctx.currentTime + i * 0.15)
        gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + i * 0.15 + 0.4)
        osc.start(ctx.currentTime + i * 0.15)
        osc.stop(ctx.currentTime + i * 0.15 + 0.4)
      })
    } catch {}
  }

  // TTS announcement
  speakAlert(text: string) {
    if (!window.speechSynthesis) return
    window.speechSynthesis.cancel()
    const utterance = new SpeechSynthesisUtterance(text)
    utterance.rate = 1.0
    utterance.pitch = 1.1
    utterance.volume = 0.9
    // Try to use a good voice
    const voices = window.speechSynthesis.getVoices()
    const preferredVoice = voices.find(v =>
      v.name.includes('Google') || v.name.includes('Microsoft') || v.lang === 'en-US'
    )
    if (preferredVoice) utterance.voice = preferredVoice
    window.speechSynthesis.speak(utterance)
  }

  // Process alert template (replace {username}, {amount} etc.)
  processTemplate(template: string, alert: StreamAlert): string {
    return template
      .replace(/{username}/g, alert.username)
      .replace(/{amount}/g, String(alert.amount || ''))
      .replace(/{currency}/g, alert.currency || '')
      .replace(/{message}/g, alert.message || '')
  }

  // Main alert trigger function
  async triggerAlert(type: AlertType, data: Partial<StreamAlert>) {
    const alert: StreamAlert = {
      id: `${Date.now()}-${Math.random().toString(36).slice(2)}`,
      type,
      username: data.username || 'Anonymous',
      message: data.message,
      amount: data.amount,
      currency: data.currency,
      timestamp: new Date()
    }

    this.alertQueue.push(alert)
    if (!this.isProcessing) this.processQueue()
  }

  private async processQueue() {
    this.isProcessing = true

    while (this.alertQueue.length > 0) {
      const alert = this.alertQueue.shift()!
      const config = this.configs.find(c => c.type === alert.type)
      if (!config?.enabled) continue

      // Play sound
      await this.playSound(alert.type)

      // Show overlay
      if (this.overlayCallback) {
        this.overlayCallback(alert, config)
      }

      // TTS
      if (config.ttsEnabled && config.ttsText) {
        const ttsText = this.processTemplate(config.ttsText, alert)
        setTimeout(() => this.speakAlert(ttsText), 500) // delay TTS slightly after sound
      }

      // Wait for overlay duration before next alert
      await new Promise(resolve => setTimeout(resolve, (config.overlayDuration + 1) * 1000))
    }

    this.isProcessing = false
  }

  // Get config for a specific alert type
  getConfig(type: AlertType): AlertConfig | undefined {
    return this.configs.find(c => c.type === type)
  }
}

export const streamAlertService = new StreamAlertService()

function roundedRect(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
  ctx.beginPath()
  ctx.moveTo(x + r, y)
  ctx.lineTo(x + w - r, y)
  ctx.quadraticCurveTo(x + w, y, x + w, y + r)
  ctx.lineTo(x + w, y + h - r)
  ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h)
  ctx.lineTo(x + r, y + h)
  ctx.quadraticCurveTo(x, y + h, x, y + h - r)
  ctx.lineTo(x, y + r)
  ctx.quadraticCurveTo(x, y, x + r, y)
  ctx.closePath()
}

// Global keyboard shortcuts for power users
export function useKeyboardShortcuts() {
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      // Only trigger when Ctrl/Cmd + Alt are held
      if (!e.ctrlKey && !e.metaKey) return

      switch(e.key.toLowerCase()) {
        case 'g': // Ctrl+G = analyze Gold
          e.preventDefault()
          sendPromptToAlchemist('Analyze XAUUSD - full Alchemist analysis')
          navigateTo('/alchemist')
          break
        case 'e': // Ctrl+E = analyze EURUSD
          e.preventDefault()
          sendPromptToAlchemist('Analyze EURUSD - full Alchemist analysis')
          navigateTo('/alchemist')
          break
        case 'b': // Ctrl+B = analyze GBPUSD
          e.preventDefault()
          sendPromptToAlchemist('Analyze GBPUSD')
          navigateTo('/alchemist')
          break
        case ' ': // Ctrl+Space = run full analysis on current pair
          e.preventDefault()
          runFullAnalysisOnCurrentPair()
          break
        case 's': // Ctrl+S = open scanner
          e.preventDefault()
          navigateTo('/scanner')
          break
        case 'r': // Ctrl+R = refresh all prices
          e.preventDefault()
          refreshAllPrices()
          break
        case 't': // Ctrl+T = open telegram bot
          e.preventDefault()
          navigateTo('/telegram-bot')
          break
        case 'm': // Ctrl+M = toggle music
          e.preventDefault()
          toggleMusic()
          break
        case 'f': // Ctrl+F = focus mode toggle
          e.preventDefault()
          toggleFocusMode()
          break
      }
    }

    window.addEventListener('keydown', handleKeyDown)
    return () => window.removeEventListener('keydown', handleKeyDown)
  }, [])
}
