```typescript
import { createContext, useContext, useRef, useState, useEffect, ReactNode } from 'react'

interface CameraFilters {
  brightness: number
  contrast: number
  saturation: number
  zoom: number
  flipped: boolean
}

interface CameraContextType {
  stream: MediaStream | null
  screenStream: MediaStream | null
  devices: MediaDeviceInfo[]
  selectedDevice: string
  error: string
  filters: CameraFilters
  isActive: boolean
  startCamera: (deviceId?: string) => Promise<void>
  stopCamera: () => void
  startScreenShare: () => Promise<void>
  stopScreenShare: () => void
  setSelectedDevice: (id: string) => void
  setFilters: (f: Partial<CameraFilters>) => void
}

const CameraContext = createContext<CameraContextType | null>(null)

export function CameraProvider({ children }: { children: ReactNode }) {
  const [stream, setStream] = useState<MediaStream | null>(null)
  const [screenStream, setScreenStream] = useState<MediaStream | null>(null)
  const [devices, setDevices] = useState<MediaDeviceInfo[]>([])
  const [selectedDevice, setSelectedDevice] = useState<string>('')
  const [error, setError] = useState<string>('')
  const [isActive, setIsActive] = useState(false)
  const [filters, setFiltersState] = useState<CameraFilters>({
    brightness: 100, contrast: 100, saturation: 100, zoom: 1, flipped: false
  })

  useEffect(() => {
    navigator.mediaDevices.enumerateDevices()
      .then(all => setDevices(all.filter(d => d.kind === 'videoinput')))
      .catch(() => {})
  }, [])

  const startCamera = async (deviceId?: string) => {
    try {
      if (stream) stream.getTracks().forEach(t => t.stop())
      const constraints: MediaStreamConstraints = {
        video: deviceId
          ? { deviceId: { exact: deviceId }, width: 1280, height: 720 }
          : { width: 1280, height: 720 },
        audio: true
      }
      const s = await navigator.mediaDevices.getUserMedia(constraints)
      setStream(s)
      setIsActive(true)
      setError('')
      // Re-enumerate after permission granted to get labels
      const all = await navigator.mediaDevices.enumerateDevices()
      setDevices(all.filter(d => d.kind === 'videoinput'))
    } catch (e: any) {
      setIsActive(false)
      if (e.name === 'NotAllowedError' || e.name === 'PermissionDeniedError') {
        setError('Camera permission denied. Please allow camera access in your browser settings and refresh.')
      } else if (e.name === 'NotFoundError') {
        setError('No camera detected. Please connect a camera and try again.')
      } else {
        setError(`Camera error: ${e.message}`)
      }
    }
  }

  const stopCamera = () => {
    stream?.getTracks().forEach(t => t.stop())
    setStream(null)
    setIsActive(false)
  }

  const startScreenShare = async () => {
    try {
      const s = await navigator.mediaDevices.getDisplayMedia({
        video: { frameRate: 30 },
        audio: true
      })
      s.getVideoTracks()[0].onended = () => setScreenStream(null)
      setScreenStream(s)
    } catch (e: any) {
      if (e.name !== 'NotAllowedError') {
        setError(`Screen share error: ${e.message}`)
      }
    }
  }

  const stopScreenShare = () => {
    screenStream?.getTracks().forEach(t => t.stop())
    setScreenStream(null)
  }

  const setFilters = (f: Partial<CameraFilters>) => {
    setFiltersState(prev => ({ ...prev, ...f }))
  }

  return (
    <CameraContext.Provider value={{
      stream, screenStream, devices, selectedDevice, error, filters,
      isActive, startCamera, stopCamera, startScreenShare, stopScreenShare,
      setSelectedDevice, setFilters
    }}>
      {children}
    </CameraContext.Provider>
  )
}

export const useCamera = () => {
  const ctx = useContext(CameraContext)
  if (!ctx) throw new Error('useCamera must be used within CameraProvider')
  return ctx
}
```

```typescript
import { useState, useEffect } from 'react'

export interface ForexEvent {
  title: string
  date: string
  impact: 'High' | 'Medium' | 'Low'
  currency: string
  forecast: string
  previous: string
  actual: string
}

export function useEconomicCalendar() {
  const [events, setEvents] = useState<ForexEvent[]>([])
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState('')

  const fetchEvents = async () => {
    try {
      const res = await fetch('https://nfs.faireconomy.media/ff_calendar_thisweek.json')
      if (!res.ok) throw new Error('Failed to fetch')
      const data = await res.json()
      const sorted = data.sort((a: ForexEvent, b: ForexEvent) =>
        new Date(a.date).getTime() - new Date(b.date).getTime()
      )
      setEvents(sorted)
      setError('')
    } catch (e) {
      setError('Calendar unavailable. Retrying...')
    } finally {
      setLoading(false)
    }
  }

  useEffect(() => {
    fetchEvents()
    const interval = setInterval(fetchEvents, 30 * 60 * 1000)
    return () => clearInterval(interval)
  }, [])

  return { events, loading, error, refetch: fetchEvents }
}

// Countdown helper — update every second in component:
export function getCountdown(dateStr: string): string {
  const diff = new Date(dateStr).getTime() - Date.now()
  if (diff <= 0) return 'Released'
  const h = Math.floor(diff / 3600000)
  const m = Math.floor((diff % 3600000) / 60000)
  const s = Math.floor((diff % 60000) / 1000)
  if (h > 24) {
    const days = Math.floor(h / 24)
    return `${days}d ${h % 24}h`
  }
  return `${h}h ${m}m ${s}s`
}
```

```typescript
// src/hooks/usePriceData.ts
const PAIRS = [
  { symbol: 'EURUSD', frankfurter: 'EUR' },
  { symbol: 'GBPUSD', frankfurter: 'GBP' },
  { symbol: 'USDJPY', frankfurter: 'JPY', invert: true },
  { symbol: 'USDCHF', frankfurter: 'CHF', invert: true },
  { symbol: 'AUDUSD', frankfurter: 'AUD' },
  { symbol: 'NZDUSD', frankfurter: 'NZD' },
  { symbol: 'USDCAD', frankfurter: 'CAD', invert: true },
]

export function usePriceData() {
  const [prices, setPrices] = useState<Record<string, { price: number; change: number; direction: 'up' | 'down' | 'flat' }>>({})
  const prevPrices = useRef<Record<string, number>>({})

  const fetchPrices = async () => {
    try {
      const res = await fetch('https://api.frankfurter.app/latest?from=USD')
      const data = await res.json()

      const updated: typeof prices = {}
      PAIRS.forEach(({ symbol, frankfurter, invert }) => {
        const raw = data.rates[frankfurter]
        const price = invert ? raw : 1 / raw  // EURUSD = 1/rate if USD is base
        // Actually for EUR: EUR/USD = 1/USD per EUR — adjust logic per pair
        const prev = prevPrices.current[symbol] || price
        const change = ((price - prev) / prev) * 100
        updated[symbol] = {
          price,
          change,
          direction: price > prev ? 'up' : price < prev ? 'down' : 'flat'
        }
        prevPrices.current[symbol] = price
      })

      setPrices(updated)
    } catch (e) {
      console.warn('Price fetch failed:', e)
    }
  }

  useEffect(() => {
    fetchPrices()
    const interval = setInterval(fetchPrices, 5000)
    return () => clearInterval(interval)
  }, [])

  return prices
}
```

```typescript
// src/hooks/useAmbientMusic.ts
const TRACKS = [
  { name: 'Ambient Focus', url: 'https://cdn.pixabay.com/audio/2022/10/16/audio_05f67b3be9.mp3' },
  { name: 'Deep Concentration', url: 'https://cdn.pixabay.com/audio/2022/03/15/audio_c8e0a5c27e.mp3' }
]

export function useAmbientMusic() {
  const audioRef = useRef<HTMLAudioElement | null>(null)
  const [playing, setPlaying] = useState(false)
  const [volume, setVolumeState] = useState(0.3)
  const [trackIndex, setTrackIndex] = useState(0)
  const [trackName, setTrackName] = useState(TRACKS[0].name)

  useEffect(() => {
    const audio = new Audio(TRACKS[0].url)
    audio.loop = true
    audio.volume = 0.3
    audioRef.current = audio

    audio.onerror = () => {
      // Fallback: Web Audio API ambient tone
      const ctx = new AudioContext()
      const freqs = [110, 165, 220, 330, 440]
      freqs.forEach((freq, i) => {
        const osc = ctx.createOscillator()
        const gain = ctx.createGain()
        osc.frequency.value = freq
        osc.type = i % 2 === 0 ? 'sine' : 'triangle'
        gain.gain.value = 0.03
        osc.connect(gain)
        gain.connect(ctx.destination)
        osc.start()
      })
      setTrackName('Ambient Tone (Generated)')
    }

    return () => { audio.pause(); audio.src = '' }
  }, [])

  const toggle = () => {
    if (!audioRef.current) return
    if (playing) {
      audioRef.current.pause()
      setPlaying(false)
    } else {
      audioRef.current.play().then(() => setPlaying(true)).catch(() => setPlaying(false))
    }
  }

  const setVolume = (v: number) => {
    if (audioRef.current) audioRef.current.volume = v
    setVolumeState(v)
  }

  const nextTrack = () => {
    const next = (trackIndex + 1) % TRACKS.length
    if (audioRef.current) {
      audioRef.current.src = TRACKS[next].url
      if (playing) audioRef.current.play()
    }
    setTrackIndex(next)
    setTrackName(TRACKS[next].name)
  }

  return { playing, toggle, volume, setVolume, trackName, nextTrack }
}
```

```typescript
// src/hooks/useSettings.ts
export function useSettings() {
  const apply = (key: string, value: string) => {
    localStorage.setItem(`jjnexus_${key}`, value)
    switch (key) {
      case 'theme':
        document.documentElement.setAttribute('data-theme', value)
        break
      case 'accent':
        document.documentElement.style.setProperty('--color-accent', value)
        break
      case 'fontSize':
        document.documentElement.style.fontSize = value + 'px'
        break
    }
  }

  // Restore all settings on app load
  const restore = () => {
    const theme = localStorage.getItem('jjnexus_theme') || 'dark'
    const accent = localStorage.getItem('jjnexus_accent') || '#D4AF37'
    const fontSize = localStorage.getItem('jjnexus_fontSize') || '16'
    apply('theme', theme)
    apply('accent', accent)
    apply('fontSize', fontSize)
  }

  return { apply, restore }
}
```
