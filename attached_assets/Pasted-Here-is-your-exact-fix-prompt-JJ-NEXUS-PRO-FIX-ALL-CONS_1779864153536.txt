// Universal proxy function
function proxyRequest(targetUrl, res) {
  const url = new URL(targetUrl)
  const protocol = url.protocol === 'https:' ? https : http

  const options = {
    hostname: url.hostname,
    path: url.pathname + url.search,
    method: 'GET',
    headers: {
      'User-Agent': 'Mozilla/5.0 (compatible; JJNexusPro/1.0)',
      'Accept': 'application/json',
      'Accept-Encoding': 'identity'
    },
    timeout: 10000
  }

  const proxyReq = protocol.request(options, (proxyRes) => {
    let data = ''
    proxyRes.on('data', chunk => data += chunk)
    proxyRes.on('end', () => {
      res.writeHead(200, {
        'Content-Type': 'application/json',
        'Access-Control-Allow-Origin': '*',
        'Access-Control-Allow-Methods': 'GET, OPTIONS',
        'Cache-Control': 'no-cache'
      })
      res.end(data)
    })
  })

  proxyReq.on('error', (err) => {
    res.writeHead(502, { 'Access-Control-Allow-Origin': '*' })
    res.end(JSON.stringify({ error: err.message }))
  })

  proxyReq.on('timeout', () => {
    proxyReq.destroy()
    res.writeHead(504, { 'Access-Control-Allow-Origin': '*' })
    res.end(JSON.stringify({ error: 'Request timeout' }))
  })

  proxyReq.end()
}

  const host = window.location.hostname

async connectToRelay(): Promise<void> {
  return new Promise((resolve, reject) => {
    // Try multiple WebSocket URLs in order
    const urls = [
      this.getRelayUrl(),
      `wss://${window.location.hostname.replace(/^(.*?)(-\d+-)(.*?)\.(.*)$/, '$1$2$3.$4')}:3001`,
      `wss://${window.location.host.split(':')[0]}:3001`,
    ]

    console.log('Trying relay WebSocket URLs:', urls)

    let tried = 0
    const tryConnect = (url: string) => {
      console.log(`Trying WebSocket: ${url}`)
      const ws = new WebSocket(url)

      const timeout = setTimeout(() => {
        ws.close()
        tried++
        if (tried < urls.length) {
          tryConnect(urls[tried])
        } else {
          reject(new Error(`Cannot reach relay server. Make sure it is running on port 3001. Tried: ${urls.join(', ')}`))
        }
      }, 5000)

      ws.onopen = () => {
        clearTimeout(timeout)
        console.log(`✅ WebSocket connected: ${url}`)
        this.ws = ws
        this.setupWebSocketHandlers(resolve, reject)
      }

      ws.onerror = () => {
        clearTimeout(timeout)
        ws.close()
        tried++
        if (tried < urls.length) {
          console.log(`Failed ${url}, trying next...`)
          tryConnect(urls[tried])
        } else {
          reject(new Error('All WebSocket connection attempts failed'))
        }
      }
    }

    tryConnect(urls[0])
  })
}

private setupWebSocketHandlers(resolve: () => void, reject: (e: Error) => void) {
  if (!this.ws) return

  this.ws.onmessage = (event) => {
    try {
      const msg = JSON.parse(event.data)
      switch (msg.type) {
        case 'connected':
          this.sessionId = msg.sessionId
          console.log('✅ Relay session:', this.sessionId)
          resolve()
          break
        case 'stream_starting':
          this.updateStats({ status: 'live', startTime: new Date() })
          this.startDurationTimer()
          break
        case 'ffmpeg_progress':
          this.updateStats({
            ffmpegFps: msg.fps,
            ffmpegBitrate: msg.bitrate,
            ffmpegSpeed: msg.speed
          })
          break
        case 'error':
          this.updateStats({ status: 'error', error: msg.message })
          break
        case 'stream_ended':
          this.updateStats({ status: 'stopped' })
          break
      }
    } catch {}
  }

  this.ws.onclose = () => {
    if (this.stats.status === 'live') {
      this.updateStats({ status: 'stopped' })
    }
  }

  this.ws.onerror = (e) => {
    console.error('WebSocket error after connection:', e)
  }
}
