// Source 1: Frankfurter API (forex pairs only, completely free)
async function fetchFrankfurter(): Promise<Record<string, number>> {
  const res = await fetch('https://api.frankfurter.app/latest?from=USD')
  const data = await res.json()
  // data.rates: { EUR: 0.92, GBP: 0.79, JPY: 149.5, ... }
  // Convert to standard format: EURUSD = 1/rate_if_USD_base
  const prices: Record<string, number> = {}
  if (data.rates) {
    // For EURUSD: EUR per USD = data.rates.EUR, but we need USD per EUR
    prices['EURUSD'] = parseFloat((1 / data.rates.EUR).toFixed(5))
    prices['GBPUSD'] = parseFloat((1 / data.rates.GBP).toFixed(5))
    prices['USDJPY'] = parseFloat(data.rates.JPY.toFixed(3))
    prices['USDCHF'] = parseFloat(data.rates.CHF.toFixed(5))
    prices['AUDUSD'] = parseFloat((1 / data.rates.AUD).toFixed(5))
    prices['NZDUSD'] = parseFloat((1 / data.rates.NZD).toFixed(5))
    prices['USDCAD'] = parseFloat(data.rates.CAD.toFixed(5))
    prices['EURGBP'] = parseFloat((data.rates.GBP / data.rates.EUR).toFixed(5))
    prices['EURJPY'] = parseFloat((data.rates.JPY / data.rates.EUR).toFixed(3))
    prices['GBPJPY'] = parseFloat((data.rates.JPY / data.rates.GBP).toFixed(3))
  }
  return prices
}

export function useLivePrices() {
  const [prices, setPrices] = useState<Record<string, number>>({})
  const [lastUpdate, setLastUpdate] = useState<Date | null>(null)
  const prevPrices = useRef<Record<string, number>>({})

  const refresh = async () => {
    const fresh = await fetchAllPrices()
    setPrices(fresh)
    setLastUpdate(new Date())
    prevPrices.current = fresh
  }

  useEffect(() => {
    refresh()
    const interval = setInterval(refresh, 10000) // every 10 seconds
    return () => clearInterval(interval)
  }, [])

  return { prices, lastUpdate, refresh }
}

const sendToTelegram = async (botToken: string, chatId: string, analysis: string, screenshotBlob?: Blob) => {
  const baseUrl = `https://api.telegram.org/bot${botToken}`

  // If screenshot exists, send as photo with caption
  if (screenshotBlob) {
    const formData = new FormData()
    formData.append('chat_id', chatId)
    formData.append('photo', screenshotBlob, 'chart.png')
    formData.append('caption', analysis.slice(0, 1024)) // Telegram caption limit
    formData.append('parse_mode', 'HTML')

    const photoRes = await fetch(`${baseUrl}/sendPhoto`, { method: 'POST', body: formData })
    if (!photoRes.ok) throw new Error('Failed to send photo')

    // If analysis is longer than 1024 chars, send rest as follow-up message
    if (analysis.length > 1024) {
      await fetch(`${baseUrl}/sendMessage`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          chat_id: chatId,
          text: '📊 <b>Full Analysis:</b>\n\n' + analysis.slice(1024),
          parse_mode: 'HTML'
        })
      })
    }
  } else {
    // Text only
    await fetch(`${baseUrl}/sendMessage`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        chat_id: chatId,
        text: analysis,
        parse_mode: 'HTML'
      })
    })
  }
}
