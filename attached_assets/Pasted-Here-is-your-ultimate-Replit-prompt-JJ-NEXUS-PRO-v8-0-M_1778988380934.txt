export async function fetchAllRealPrices(): Promise<Record<string, number>> {
  const prices: Record<string, number> = {}
  const errors: string[] = []

  // FOREX from Frankfurter
  try {
    const res = await fetch(`${FRANKFURTER_BASE}/latest?from=USD`)
    if (!res.ok) throw new Error(`Frankfurter HTTP ${res.status}`)
    const data = await res.json()
    const r = data.rates
    if (!r) throw new Error('No rates in response')

    prices.EURUSD = parseFloat((1 / r.EUR).toFixed(5))
    prices.GBPUSD = parseFloat((1 / r.GBP).toFixed(5))
    prices.USDJPY = parseFloat(r.JPY.toFixed(3))
    prices.USDCHF = parseFloat(r.CHF.toFixed(5))
    prices.AUDUSD = parseFloat((1 / r.AUD).toFixed(5))
    prices.NZDUSD = parseFloat((1 / r.NZD).toFixed(5))
    prices.USDCAD = parseFloat(r.CAD.toFixed(5))
    if (r.GBP && r.EUR) prices.EURGBP = parseFloat((r.GBP / r.EUR).toFixed(5))
    if (r.JPY && r.EUR) prices.EURJPY = parseFloat((r.JPY / r.EUR).toFixed(3))
    if (r.JPY && r.GBP) prices.GBPJPY = parseFloat((r.JPY / r.GBP).toFixed(3))
    if (r.CHF && r.EUR) prices.EURCHF = parseFloat((r.CHF / r.EUR).toFixed(5))
    if (r.JPY && r.AUD) prices.AUDJPY = parseFloat((r.JPY / r.AUD).toFixed(3))
    if (r.JPY && r.CAD) prices.CADJPY = parseFloat((r.JPY / r.CAD).toFixed(3))
    if (r.CAD && r.AUD) prices.AUDCAD = parseFloat((r.CAD / r.AUD).toFixed(5))
    if (r.NZD && r.AUD) prices.AUDNZD = parseFloat((r.NZD / r.AUD).toFixed(5))
    if (r.MXN) prices.USDMXN = parseFloat(r.MXN.toFixed(3))
    if (r.ZAR) prices.USDZAR = parseFloat(r.ZAR.toFixed(3))
    if (r.NOK) prices.USDNOK = parseFloat(r.NOK.toFixed(3))
    if (r.SEK) prices.USDSEK = parseFloat(r.SEK.toFixed(3))
    if (r.TRY) prices.USDTRY = parseFloat(r.TRY.toFixed(3))
    if (r.SGD) prices.USDSGD = parseFloat(r.SGD.toFixed(5))
    console.log('✅ Forex prices loaded from Frankfurter')
  } catch (e: any) {
    errors.push(`Frankfurter: ${e.message}`)
    console.error('❌ Frankfurter failed:', e.message)
  }

  // GOLD from metals.live
  try {
    const res = await fetch(METALS_LIVE, { signal: AbortSignal.timeout(8000) })
    if (!res.ok) throw new Error(`metals.live HTTP ${res.status}`)
    const data = await res.json()
    if (!Array.isArray(data)) throw new Error('Invalid metals.live response')

    const gold = data.find((m: any) =>
      m.metal?.toLowerCase() === 'gold' || m.symbol === 'XAU' || m.name?.toLowerCase().includes('gold')
    )
    if (gold && gold.price && gold.price > 1000 && gold.price < 10000) {
      prices.XAUUSD = parseFloat(gold.price.toFixed(2))
      prices.GOLD = prices.XAUUSD
      console.log(`✅ Gold price: $${prices.XAUUSD} from metals.live`)
    } else {
      throw new Error('Gold price out of valid range or not found')
    }

    const silver = data.find((m: any) =>
      m.metal?.toLowerCase() === 'silver' || m.symbol === 'XAG'
    )
    if (silver?.price && silver.price > 5 && silver.price < 500) {
      prices.XAGUSD = parseFloat(silver.price.toFixed(3))
    }
  } catch (e: any) {
    errors.push(`metals.live: ${e.message}`)
    console.warn('⚠️ metals.live failed, trying fallback for Gold...')

    // Fallback: exchangerate.host XAU
    try {
      const res = await fetch(XAU_FALLBACK, { signal: AbortSignal.timeout(8000) })
      const data = await res.json()
      if (data?.result && data.result > 1000 && data.result < 10000) {
        prices.XAUUSD = parseFloat(data.result.toFixed(2))
        prices.GOLD = prices.XAUUSD
        console.log(`✅ Gold fallback: $${prices.XAUUSD} from exchangerate.host`)
      }
    } catch (e2: any) {
      console.error('❌ All Gold price sources failed')
      errors.push(`Gold fallback: ${e2.message}`)
    }
  }

  // CRYPTO from CoinGecko
  try {
    const res = await fetch(
      `${COINGECKO}/simple/price?ids=bitcoin,ethereum,solana,ripple,binancecoin,dogecoin&vs_currencies=usd&include_24hr_change=true`,
      { signal: AbortSignal.timeout(10000) }
    )
    if (!res.ok) throw new Error(`CoinGecko HTTP ${res.status}`)
    const data = await res.json()
    if (data.bitcoin?.usd) prices.BTCUSD = data.bitcoin.usd
    if (data.ethereum?.usd) prices.ETHUSD = data.ethereum.usd
    if (data.solana?.usd) prices.SOLUSD = data.solana.usd
    if (data.ripple?.usd) prices.XRPUSD = data.ripple.usd
    if (data.binancecoin?.usd) prices.BNBUSD = data.binancecoin.usd
    if (data.dogecoin?.usd) prices.DOGEUSD = data.dogecoin.usd
    console.log('✅ Crypto prices from CoinGecko')
  } catch (e: any) {
    errors.push(`CoinGecko: ${e.message}`)
  }

  // INDICES from Yahoo Finance via proxy
  const yahooSymbols: Record<string, string> = {
    US30: '%5EDJI', NAS100: '%5EIXIC', SPX500: '%5EGSPC',
    GER40: '%5EGDAXI', UK100: '%5EFTSE', JP225: '%5EN225',
    AUS200: '%5EAXJO', VIX: '%5EVIX', DXY: 'DX-Y.NYB',
    USOIL: 'CL%3DF', GCF: 'GC%3DF'
  }

  await Promise.allSettled(
    Object.entries(yahooSymbols).map(async ([name, sym]) => {
      try {
        const url = `https://query1.finance.yahoo.com/v8/finance/chart/${sym}?interval=1m&range=1d`
        const proxyUrl = `https://api.allorigins.win/get?url=${encodeURIComponent(url)}`
        const res = await fetch(proxyUrl, { signal: AbortSignal.timeout(12000) })
        const wrapper = await res.json()
        const parsed = JSON.parse(wrapper.contents)
        const price = parsed?.chart?.result?.[0]?.meta?.regularMarketPrice
        if (price && price > 0) {
          prices[name] = parseFloat(price.toFixed(name === 'DXY' ? 3 : 2))
          // Use Yahoo Gold as additional validation
          if (name === 'GCF' && price > 1000 && price < 10000 && !prices.XAUUSD) {
            prices.XAUUSD = parseFloat(price.toFixed(2))
          }
        }
      } catch {}
    })
  )

  // Log any Gold price for debugging
  if (prices.XAUUSD) {
    console.log(`💰 FINAL GOLD PRICE: $${prices.XAUUSD}`)
  } else {
    console.error('❌ GOLD PRICE UNAVAILABLE — all sources failed')
  }

  // Sanity checks — reject impossible values
  const SANITY: Record<string, [number, number]> = {
    XAUUSD: [2000, 5000], EURUSD: [0.8, 1.5], GBPUSD: [1.0, 1.8],
    USDJPY: [100, 200], BTCUSD: [10000, 500000], ETHUSD: [500, 20000],
    US30: [20000, 60000], NAS100: [10000, 30000], DXY: [80, 120],
  }

  Object.entries(SANITY).forEach(([pair, [min, max]]) => {
    if (prices[pair] && (prices[pair] < min || prices[pair] > max)) {
      console.error(`❌ SANITY FAIL: ${pair} = ${prices[pair]} (expected ${min}-${max}) — REMOVING`)
      delete prices[pair]
    }
  })

  if (errors.length > 0) {
    console.warn('Price fetch errors:', errors)
  }

  return prices
}

export async function fetchRealForexNews(): Promise<NewsItem[]> {
  const allNews: NewsItem[] = []

  const sources = [
    { name: 'FXStreet', url: 'https://www.fxstreet.com/rss/news' },
    { name: 'ForexLive', url: 'https://www.forexlive.com/feed' },
    { name: 'Reuters Business', url: 'https://feeds.reuters.com/reuters/businessNews' },
    { name: 'MarketWatch', url: 'https://feeds.marketwatch.com/marketwatch/topstories' },
  ]

  await Promise.allSettled(sources.map(async source => {
    try {
      const proxyUrl = `https://api.allorigins.win/get?url=${encodeURIComponent(source.url)}`
      const res = await fetch(proxyUrl, { signal: AbortSignal.timeout(10000) })
      const wrapper = await res.json()
      const xml = wrapper.contents

      // Parse RSS XML
      const parser = new DOMParser()
      const doc = parser.parseFromString(xml, 'text/xml')
      const items = doc.querySelectorAll('item')

      items.forEach(item => {
        const title = item.querySelector('title')?.textContent?.trim() || ''
        const description = item.querySelector('description')?.textContent?.trim() || ''
        const pubDate = item.querySelector('pubDate')?.textContent?.trim() || ''
        const link = item.querySelector('link')?.textContent?.trim() || ''

        if (title && title.length > 10) {
          allNews.push({
            id: `${source.name}-${Date.now()}-${Math.random()}`,
            title: title.replace(/<[^>]*>/g, ''),
            description: description.replace(/<[^>]*>/g, '').slice(0, 300),
            source: source.name,
            publishedAt: new Date(pubDate || Date.now()),
            url: link,
            affectedPairs: extractAffectedPairs(title + ' ' + description),
            impactScore: 0, // scored by AI
            direction: 'Neutral'
          })
        }
      })
    } catch (e) {
      console.warn(`Failed to fetch ${source.name}:`, e)
    }
  }))

  // Sort by date, newest first
  return allNews.sort((a, b) => b.publishedAt.getTime() - a.publishedAt.getTime())
}

function extractAffectedPairs(text: string): string[] {
  const pairs: string[] = []
  const keywords: Record<string, string[]> = {
    'USD': ['dollar', 'fed', 'federal reserve', 'powell', 'fomc', 'nfp', 'payroll', 'cpi usa', 'us inflation'],
    'EUR': ['euro', 'ecb', 'lagarde', 'eurozone', 'european', 'germany', 'france', 'eu '],
    'GBP': ['pound', 'sterling', 'boe', 'bank of england', 'bailey', 'uk ', 'britain', 'brexit'],
    'JPY': ['yen', 'boj', 'bank of japan', 'ueda', 'japan', 'nikkei'],
    'GOLD': ['gold', 'xau', 'bullion', 'precious metal'],
    'OIL': ['oil', 'crude', 'opec', 'petroleum', 'wti', 'brent'],
  }

  const textLower = text.toLowerCase()
  Object.entries(keywords).forEach(([currency, words]) => {
    if (words.some(word => textLower.includes(word))) {
      pairs.push(currency)
    }
  })

  return [...new Set(pairs)]
}

interface FundedAccountConfig {
  firmName: string
  accountSize: number // 1000
  dailyDrawdownLimit: number // 30
  maxDrawdownLimit: number // 60
  profitTarget: number // usually 8-10% = $80-100
  challengeDuration: number // 30 days
  startDate: string
  startBalance: number
  currentBalance: number
  highWaterMark: number // highest balance reached
}

// Real-time calculations
interface AccountStatus {
  currentBalance: number
  startBalance: number
  highWaterMark: number
  totalPnL: number
  totalPnLPercent: number
  todayPnL: number
  todayPnLPercent: number
  dailyDrawdownUsed: number
  dailyDrawdownRemaining: number
  maxDrawdownUsed: number
  maxDrawdownRemaining: number
  profitTarget: number
  progressToTarget: number
  daysElapsed: number
  daysRemaining: number
  tradesRemaining: number // before daily limit
  status: 'Safe' | 'Warning' | 'Danger' | 'Failed' | 'Passed'
  canTradeToday: boolean
  reason: string // why cannot trade if applicable
}

function calculateAccountStatus(
  config: FundedAccountConfig,
  trades: JournalTrade[]
): AccountStatus {
  const today = new Date().toDateString()
  const todayTrades = trades.filter(t =>
    new Date(t.date).toDateString() === today
  )

  const todayPnL = todayTrades.reduce((sum, t) => sum + (t.pnl || 0), 0)
  const totalPnL = config.currentBalance - config.startBalance

  // Calculate from HIGH WATER MARK (most prop firms use this)
  const drawdownFromHWM = config.highWaterMark - config.currentBalance
  const maxDrawdownUsed = (drawdownFromHWM / config.accountSize) * 100

  const dailyDrawdownUsed = Math.abs(Math.min(0, todayPnL))
  const dailyDrawdownRemaining = config.dailyDrawdownLimit - dailyDrawdownUsed

  const daysElapsed = Math.floor(
    (new Date().getTime() - new Date(config.startDate).getTime()) / (1000 * 60 * 60 * 24)
  )

  let status: AccountStatus['status'] = 'Safe'
  let canTradeToday = true
  let reason = ''

  // Check if FAILED
  if (totalPnL <= -config.maxDrawdownLimit) {
    status = 'Failed'
    canTradeToday = false
    reason = `MAX DRAWDOWN BREACHED: Lost $${Math.abs(totalPnL).toFixed(2)} of $${config.maxDrawdownLimit} limit`
  } else if (dailyDrawdownUsed >= config.dailyDrawdownLimit) {
    canTradeToday = false
    reason = `DAILY DRAWDOWN REACHED: Lost $${dailyDrawdownUsed.toFixed(2)} today. Limit: $${config.dailyDrawdownLimit}. Come back tomorrow.`
    status = 'Danger'
  } else if (dailyDrawdownRemaining < config.dailyDrawdownLimit * 0.3) {
    status = 'Warning'
  } else if (totalPnL >= config.profitTarget) {
    status = 'Passed'
  }

  return {
    currentBalance: config.currentBalance,
    startBalance: config.startBalance,
    highWaterMark: config.highWaterMark,
    totalPnL,
    totalPnLPercent: (totalPnL / config.accountSize) * 100,
    todayPnL,
    todayPnLPercent: (todayPnL / config.accountSize) * 100,
    dailyDrawdownUsed,
    dailyDrawdownRemaining,
    maxDrawdownUsed,
    maxDrawdownRemaining: config.maxDrawdownLimit - Math.abs(Math.min(0, totalPnL)),
    profitTarget: config.profitTarget,
    progressToTarget: (Math.max(0, totalPnL) / config.profitTarget) * 100,
    daysElapsed,
    daysRemaining: config.challengeDuration - daysElapsed,
    tradesRemaining: 3 - todayTrades.length, // max 3 trades per day rule
    status,
    canTradeToday,
    reason
  }
}

function computeRealMetrics(trades: JournalTrade[]): PerformanceMetrics {
  const wins = trades.filter(t => (t.pnl || 0) > 0)
  const losses = trades.filter(t => (t.pnl || 0) <= 0)

  const totalPnL = trades.reduce((sum, t) => sum + (t.pnl || 0), 0)
  const grossWins = wins.reduce((sum, t) => sum + (t.pnl || 0), 0)
  const grossLosses = Math.abs(losses.reduce((sum, t) => sum + (t.pnl || 0), 0))

  // Equity curve
  let runningBalance = 1000 // start balance
  const equityCurve = trades
    .sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime())
    .map(t => {
      runningBalance += (t.pnl || 0)
      return {
        date: new Date(t.date).toLocaleDateString(),
        balance: parseFloat(runningBalance.toFixed(2)),
        drawdown: 0 // calculated separately
      }
    })

  // Win rate by pair
  const pairGroups: Record<string, JournalTrade[]> = {}
  trades.forEach(t => {
    if (!pairGroups[t.pair]) pairGroups[t.pair] = []
    pairGroups[t.pair].push(t)
  })
  const winRateByPair: Record<string, { wins: number; losses: number; winRate: number; pnl: number }> = {}
  Object.entries(pairGroups).forEach(([pair, pairTrades]) => {
    const pairWins = pairTrades.filter(t => (t.pnl || 0) > 0).length
    winRateByPair[pair] = {
      wins: pairWins,
      losses: pairTrades.length - pairWins,
      winRate: (pairWins / pairTrades.length) * 100,
      pnl: pairTrades.reduce((sum, t) => sum + (t.pnl || 0), 0)
    }
  })

  return {
    totalTrades: trades.length,
    winRate: trades.length > 0 ? (wins.length / trades.length) * 100 : 0,
    profitFactor: grossLosses > 0 ? grossWins / grossLosses : wins.length > 0 ? 999 : 0,
    totalPnL,
    grossWins,
    grossLosses,
    equityCurve,
    winRateByPair,
    bestPair: Object.entries(winRateByPair).sort(([,a],[,b]) => b.winRate - a.winRate)[0]?.[0] || 'None yet',
    worstPair: Object.entries(winRateByPair).sort(([,a],[,b]) => a.winRate - b.winRate)[0]?.[0] || 'None yet',
    // ... all other metrics computed from real data
  }
}

export async function fetchRealCOTData(): Promise<COTData[]> {

    const wrapper = await res.json()

    if (!rawData?.data?.length) {

    }

    console.log(`✅ COT data: ${rawData.data.length} rows fetched`)

    // Process the data
    const rows = rawData.data
    const results: COTData[] = []

    // Group rows by contract
    const byContract: Record<string, any[]> = {}
    rows.forEach((row: any) => {
      const contractName = row[8] || row[3] || ''
      if (!byContract[contractName]) byContract[contractName] = []
      byContract[contractName].push(row)
    })

    Object.entries(CFTC_CONTRACT_MAP).forEach(([contractName, pair]) => {
      const contractRows = byContract[contractName]
      if (!contractRows || contractRows.length < 2) return

      // Sort by date descending
      contractRows.sort((a, b) =>
        new Date(b[4] || b[3]).getTime() - new Date(a[4] || a[3]).getTime()
      )

      const latest = contractRows[0]
      const previous = contractRows[1]

      // Column indices vary by dataset version — try multiple

      const getNonCommShort = (row: any) => parseInt(row[14] || row[10] || '0')
