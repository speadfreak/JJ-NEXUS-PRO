**Formula 2 — Wang Sentiment Index (Zhang and Laws, Wang 2001)**

```typescript
// 3-year rolling lookback = 156 weekly data points
// For each trader type i in market j at week t:
// SI = (Current Net - Min of past 156 weeks) / (Max of past 156 weeks - Min of past 156 weeks)

function calculateWangSentimentIndex(
  currentNetPosition: number,
  historicalNetPositions: number[], // last 156 weeks
): number {
  const min156 = Math.min(...historicalNetPositions)
  const max156 = Math.max(...historicalNetPositions)
  if (max156 === min156) return 0.5
  const si = (currentNetPosition - min156) / (max156 - min156)
  return Math.max(0, Math.min(1, parseFloat(si.toFixed(4))))
}
// Result: 0 = 3-year extreme bearish, 1 = 3-year extreme bullish
// 0.5 = neutral, above 0.5 = relatively bullish, below 0.5 = relatively bearish
```

// EXIT SIGNAL CONDITIONS (Dreesmann)
export function generateExitSignal(
  direction: SignalDirection,
  nonCommercialIndex26: number,
  smaRising: boolean
): { shouldExit: boolean; reason: string } {
  if (direction === 'LONG') {
    // Exit long when:
    // Condition A: Non-commercial index above 0.70 (speculative crowd now extremely long)
    // Condition B: SMA turns declining
    const exitA = nonCommercialIndex26 >= 0.70
    const exitB = !smaRising
    if (exitA && exitB) return {
      shouldExit: true,
      reason: 'Both exit conditions met: Non-commercial extreme long AND SMA declining'
    }
    if (exitA) return { shouldExit: true, reason: 'Non-commercial extreme long (0.70+) — crowd too bullish' }
    if (exitB) return { shouldExit: true, reason: 'SMA turned declining — price momentum reversing' }
  }

  if (direction === 'SHORT') {
    // Exit short when:
    const exitA = nonCommercialIndex26 <= 0.30
    const exitB = smaRising
    if (exitA && exitB) return {
      shouldExit: true,
      reason: 'Both exit conditions met: Non-commercial extreme short AND SMA rising'
    }
    if (exitA) return { shouldExit: true, reason: 'Non-commercial extreme short (0.30-) — crowd too bearish' }
    if (exitB) return { shouldExit: true, reason: 'SMA turned rising — price momentum reversing' }
  }

  return { shouldExit: false, reason: 'Hold position — exit conditions not met' }
}

export async function buildCOTDataset(pair: string, currentPrice: number, historicalPrices: number[]): Promise<COTWeeklyData[]> {
  const marketInfo = CFTC_MARKET_MAP[pair]
  if (!marketInfo) return []

  if (rawData.length < 26) {
    console.warn(`Insufficient COT data for ${pair}: only ${rawData.length} weeks`)
    return []
  }

  const sortedData = rawData.sort((a: any, b: any) =>
    new Date(b.report_date_as_yyyy_mm_dd).getTime() -
    new Date(a.report_date_as_yyyy_mm_dd).getTime()
  )

  // Build weekly dataset with all calculations
  return sortedData.slice(0, 156).map((row: any, index: number) => {
    const commLong = parseInt(row.comm_positions_long_all || '0')
    const commShort = parseInt(row.comm_positions_short_all || '0')

    const nonCommLong = parseInt(row.noncomm_positions_long_all || '0')
    const nonCommShort = parseInt(row.noncomm_positions_short_all || '0')

    const nonRepNet = nonRepLong - nonRepShort

    const totalOI = parseInt(row.open_interest_all || '0')

    // 26-week net position arrays for Dreesmann indices

      parseInt(r.comm_positions_long_all || '0') - parseInt(r.comm_positions_short_all || '0')
    )
    const historical26NonComm = sortedData.slice(index, index + 26).map((r: any) =>
      parseInt(r.noncomm_positions_long_all || '0') - parseInt(r.noncomm_positions_short_all || '0')
    )
    const historical26NonRep = sortedData.slice(index, index + 26).map((r: any) =>
      parseInt(r.nonrept_positions_long_all || '0') - parseInt(r.nonrept_positions_short_all || '0')
    )

    // 156-week net position arrays for Wang indices
    const historical156Comm = sortedData.slice(index, index + 156).map((r: any) =>

    // Approximate weekly prices (use live price for most recent, interpolate for historical)

  const [lastRefresh, setLastRefresh] = useState<Date | null>(null)

  // Current week data (index 0 = most recent)
  const current = cotData[0]

  // Generate Dreesmann signal
  const signal = current ? generateCOTSignal(

    current.commercialIndex26,
    current.commercialLongOIPercent,

    current.nonReportableIndex26,
    current.smaRising,
    current.nonCommercialIndex26
  ) : null

  // Generate exit signal
  const exitSignal = current && signal ? generateExitSignal(
    signal.direction,
    current.nonCommercialIndex26,
    current.smaRising
  ) : null

  // Detect extreme positions
  const extremes = current ? detectExtremePosition(
    current.commercialSI156,
    current.nonCommercialSI156,
    current.nonReportableSI156,
    selectedPair
  ) : null

  // Load COT data when pair changes
  useEffect(() => {
    loadCOTData(selectedPair)
  }, [selectedPair])

  const loadCOTData = async (pair: string) => {
    setLoading(true)
    setError(null)
    try {
      const currentPrice = prices[pair] || 0
      const historicalPrices = JSON.parse(
        localStorage.getItem(`jjnexus_price_history_${pair}`) || '[]'
      )
      const data = await buildCOTDataset(pair, currentPrice, historicalPrices)
      if (data.length === 0) {
        setError('CFTC data unavailable. COT reports are released every Friday. Check your connection.')
      } else {
        setCotData(data)
        setLastRefresh(new Date())
      }
    } catch (e: any) {
      setError(`Failed to load COT data: ${e.message}`)
    } finally {
      setLoading(false)
    }
  }
